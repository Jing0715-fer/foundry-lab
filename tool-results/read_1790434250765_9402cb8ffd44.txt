   144→function timeAgo(iso: string): string {
   145→  const then = new Date(iso).getTime();
   146→  const s = Math.max(1, Math.round((Date.now() - then) / 1000));
   147→  if (s < 60) return `${s}s ago`;
   148→  const m = Math.round(s / 60);
   149→  if (m < 60) return `${m}m ago`;
   150→  const h = Math.round(m / 60);
   151→  return `${h}h ago`;
   152→}

   156→export function ToolsPanel() {
   157→  const toast = useAppStore((s) => s.toast);
   158→  const [scan, setScan] = React.useState<ScanResponse | null>(null);
   159→  const [loading, setLoading] = React.useState(true);
   160→  const [scanning, setScanning] = React.useState(false);
   161→  const [scanError, setScanError] = React.useState<string | null>(null);
   162→
   163→  // Install dialog state.
   164→  const [installJob, setInstallJob] = React.useState<InstallJobDTO | null>(null);
   165→  const [installOpen, setInstallOpen] = React.useState(false);
   166→  const [startingInstall, setStartingInstall] = React.useState<string | null>(null);
   167→  const pollRef = React.useRef<ReturnType<typeof setInterval> | null>(null);
   168→  const logBoxRef = React.useRef<HTMLDivElement>(null);
   169→
   170→  const runScan = React.useCallback(
   171→    async (silent = false) => {
   172→      if (!silent) setScanning(true);
   173→      try {
   174→        const res = await fetch("/api/tools/scan", { cache: "no-store" });
   175→        if (!res.ok) throw new Error(`Scan failed (${res.status})`);
   176→        const data = (await res.json()) as ScanResponse;
   177→        setScan(data);
   178→        setScanError(null);
   179→      } catch (e) {
   180→        setScanError((e as Error).message);
   181→      } finally {
   182→        setScanning(false);
   183→        setLoading(false);
   184→      }
   185→    },
   186→    [],
   187→  );
   188→
   189→  React.useEffect(() => {
   190→    void runScan();
   191→  }, [runScan]);
   192→
   193→  // If a scan arrives with an active install job, surface it in the dialog.
   194→  React.useEffect(() => {
   195→    if (scan?.activeInstall && !installOpen) {
   196→      setInstallJob(scan.activeInstall);
   197→      setInstallOpen(true);
   198→    }
   199→  }, [scan?.activeInstall, installOpen]);
   200→
   201→  // Poll the open install job until it finishes.
   202→  React.useEffect(() => {
   203→    if (!installOpen || !installJob) {
   204→      if (pollRef.current) clearInterval(pollRef.current);
   205→      return;
   206→    }
   207→    if (installJob.status !== "running") {
   208→      if (pollRef.current) clearInterval(pollRef.current);
   209→      // One final scan to refresh statuses when an install finishes.
   210→      if (pollRef.current || installJob.status === "completed") void runScan(true);
   211→      return;
   212→    }
   213→    pollRef.current = setInterval(async () => {
   214→      try {
   215→        const res = await fetch(`/api/tools/install/${installJob.id}`, {
   216→          cache: "no-store",
   217→        });
   218→        if (!res.ok) return;
   219→        const job = (await res.json()) as InstallJobDTO;
   220→        setInstallJob(job);
   221→        if (job.status !== "running") {
   222→          if (job.status === "completed") {
   223→            toast({
   224→              title: `${job.label} installed`,
   225→              description: "Re-scanning environment…",
   226→            });
   227→          }
   228→          void runScan(true);
   229→        }
   230→      } catch {
   231→        /* transient poll failure — keep trying */
   232→      }
   233→    }, 1000);
   234→    return () => {
   235→      if (pollRef.current) clearInterval(pollRef.current);
   236→    };
   237→  }, [installOpen, installJob, runScan, toast]);
   238→
   239→  // Auto-scroll the install log.
   240→  React.useEffect(() => {
   241→    if (logBoxRef.current) {
   242→      logBoxRef.current.scrollTop = logBoxRef.current.scrollHeight;
   243→    }
   244→  }, [installJob?.logs.length]);
   245→
   246→  const startInstall = React.useCallback(
   247→    async (key: string, label: string) => {
   248→      setStartingInstall(key);
   249→      try {
   250→        const res = await fetch("/api/tools/install", {
   251→          method: "POST",
   252→          headers: { "Content-Type": "application/json" },
   253→          body: JSON.stringify({ key }),
   254→        });
   255→        const data = await res.json();
   256→        if (!res.ok) {
   257→          toast({
   258→            title: "Cannot start install",
   259→            description: data.error ?? "Unknown error",
   260→            variant: "destructive",
   261→          });
   262→          return;
   263→        }
   264→        setInstallJob(data as InstallJobDTO);
   265→        setInstallOpen(true);
   266→      } catch (e) {
   267→        toast({
   268→          title: "Install request failed",
   269→          description: (e as Error).message,
   270→          variant: "destructive",
   271→        });
   272→      } finally {
   273→        setStartingInstall(null);
   274→      }
   275→    },
   276→    [toast],
   277→  );
   278→
   279→  if (loading) {
   280→    return (
   281→      <div className="p-6">
   282→        <PanelSkeleton count={4} />
   283→      </div>
   284→    );
   285→  }
   286→
   287→  const summary = scan?.summary;
   288→
   289→  return (
   290→    <div className="mx-auto max-w-5xl space-y-6 p-4 sm:p-6">
   291→      {/* ── Header ───────────────────────────────────────────────────────── */}
   292→      <div className="flex flex-wrap items-start justify-between gap-3">
   293→        <div className="space-y-1">
   294→          <div className="flex items-center gap-2">
   295→            <Wrench className="size-5 text-primary" />
   296→            <h2 className="text-lg font-semibold tracking-tight">
   297→              Environment &amp; Toolchain
   298→            </h2>
   299→          </div>
   300→          <p className="text-sm text-muted-foreground">
   301→            Installation status for every dependency the studio uses — runtime,
   302→            built-in real algorithm engines, and external research tools.
   303→            Uninstalled items with a{" "}
   304→            <span className="font-medium text-foreground">Install</span> button
   305→            support one-click installation.
   306→          </p>
   307→        </div>
   308→        <Button
   309→          variant="outline"
   310→          size="sm"
   311→          onClick={() => void runScan()}
   312→          disabled={scanning}
   313→        >
   314→          {scanning ? (
   315→            <Loader2 className="size-3.5 animate-spin" />
   316→          ) : (
   317→            <RefreshCw className="size-3.5" />
   318→          )}
   319→          Re-scan
   320→        </Button>
   321→      </div>
   322→
   323→      {scanError && (
   324→        <Card className="border-destructive/40">
   325→          <CardContent className="flex items-center gap-3 p-4 text-sm">

   328→              <p className="font-medium">Environment scan failed</p>
   329→              <p className="text-muted-foreground">{scanError}</p>
   330→            </div>
   331→            <Button size="sm" variant="outline" onClick={() => void runScan()}>
   332→              Retry
   333→            </Button>
   334→          </CardContent>
   335→        </Card>
   336→      )}
   337→
   338→      {/* ── Summary cards ────────────────────────────────────────────────── */}
   339→      {scan && summary && (

   341→          <SummaryCard

   344→            value={`${summary.runtime.installed}/${summary.runtime.total}`}
   345→            caption={summary.runtime.coreReady ? "core ready" : "python3 + numpy required"}
   346→            ok={summary.runtime.coreReady}
   347→          />
   348→          <SummaryCard
   349→            icon={<Boxes className="size-4" />}
   350→            title="Real algorithm engines"
   351→            value={`${summary.engines.ok}/${summary.engines.total}`}
   352→            caption={summary.engines.ok === summary.engines.total ? "all self-tests pass" : "self-test failures"}
   353→            ok={summary.engines.ok === summary.engines.total}
   354→          />
   355→          <SummaryCard
   356→            icon={<PackageCheck className="size-4" />}

   365→      {/* ── ① Runtime dependencies ───────────────────────────────────────── */}

   387→
   388→      {/* ── ② Built-in engines ──────────────────────────────────────────── */}

   392→            id="engines-heading"

   394→            title="Built-in Real Algorithm Engines"

   397→          />

   468→                  }
   469→                >
   470→                  {installJob.status}
   471→                </Badge>
   472→              )}

   474→            <DialogDescription className="font-mono text-[11px] break-all">
   475→              {installJob?.command}
   476→            </DialogDescription>
   477→          </DialogHeader>
   478→          {installJob && (
   479→            <div className="space-y-3">
   480→              {installJob.status === "running" && (
   481→                <Progress value={100} className="h-1 animate-pulse" />
   482→              )}
   483→              <div
   484→                ref={logBoxRef}
   485→                className="max-h-[50vh] overflow-y-auto rounded-md border bg-zinc-950 p-3 font-mono text-[11px] leading-relaxed text-zinc-200"
   486→                aria-live="polite"
   487→              >

   490→                    key={i}

   504→                    <Loader2 className="mr-1 inline size-3 animate-spin" />

   506→                  </div>

   508→              </div>
   509→              <div className="flex justify-between text-[11px] text-muted-foreground">

   522→            </div>

   524→        </DialogContent>
   525→      </Dialog>
   526→    </div>

   528→}

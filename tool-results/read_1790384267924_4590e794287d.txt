  1770→function relionLabelKey(s: string): string {
  1771→  return s.toLowerCase().replace(/[?:*]/g, "").replace(/\s+/g, " ").trim();
  1772→}

  1774→/** First ~2 sentences of a RELION helptext, capped for tooltip use. */
  1775→function relionHint(help: string): string | undefined {
  1776→  const clean = help.replace(/\s+/g, " ").trim();
  1777→  if (!clean) return undefined;
  1778→  const sentences = clean.match(/[^.!?]+[.!?]+/g) ?? [clean];
  1779→  let out = "";
  1780→  for (const s of sentences) {
  1781→    if (out && (out + s).length > 260) break;
  1782→    out += s + " ";
  1783→    if (out.length > 260) break;
  1784→  }
  1785→  out = out.trim() || clean.slice(0, 260);
  1786→  return out.length > 300 ? out.slice(0, 297) + "…" : out;
  1787→}

  1789→function applyRelionParamTable(): void {
  1790→  for (const spec of JOB_TYPES) {
  1791→    const table = RELION_OPTIONS[spec.key];
  1792→    if (!table) continue;
  1793→    const isTomo = spec.group === "Tomography";
  1794→    const skipWorld = isTomo ? "spa" : "tomo";
  1795→    // t386 — alias values may be ARRAYS (one curated composite knob owning
  1796→    // several raw options, e.g. class2d's algorithm → do_em + do_grad), so
  1797→    // the skip-set flattens before use
  1798→    const aliased = new Set(
  1799→      Object.values(RELION_ALIASES[spec.key] ?? {}).flatMap((v) =>
  1800→        Array.isArray(v) ? v : [v]
  1801→      )
  1802→    );
  1803→    // t387 — the audit-verified inert controls (RELION_DROP: no builder
  1804→    // reads them, no CLI flag to ride, no dispatch gate consults them) and
  1805→    // the GPU dialog keys (the submit dialog owns the GPU decision — the
  1806→    // form's use_gpu/gpu_ids were dead twins of the run dialog's stepper)
  1807→    const dropped = new Set([
  1808→      ...(RELION_DROP[spec.key] ?? []),
  1809→      ...GPU_DIALOG_KEYS,
  1810→    ]);
  1811→    const have = new Set(spec.params.map((p) => p.key));
  1812→    const labelTwin = new Set(spec.params.map((p) => relionLabelKey(p.label)));
  1813→
  1814→    const relionTabOrder: string[] = [];
  1815→    for (const tab of table.tabs) {
  1816→      let populated = spec.params.some((p) => p.tab === tab.label);
  1817→      for (const o of tab.options) {
  1818→        if (o.world === skipWorld) continue;
  1819→        if (have.has(o.key) || aliased.has(o.key) || dropped.has(o.key)) continue;
  1820→        const def = table.options[o.key];
  1821→        if (!def) continue;
  1822→        // rule 3: same label under a curated name — one knob, one control
  1823→        if (labelTwin.has(relionLabelKey(def.label))) continue;
  1824→        // rule 4: RELION "node" options (the pipeline input STAR pickers —
  1825→        // fn_img/fn_mic/fn_*_star …) are the EDGE-wired inputs in cryoflow;
  1826→        // fn_cont belongs to cryoflow's own resume system. Everything else
  1827→        // named fn_* is a real user-picked file (MTF curve, gain ref …) and
  1828→        // stays as a path param.
  1829→        if (
  1830→          def.type === "path" &&
  1831→          def.key.startsWith("fn_") &&
  1832→          (def.key.endsWith("_star") ||
  1833→            ["fn_img", "fn_mic", "fn_mics", "fn_movies", "fn_ref", "fn_cont", "fn_mask", "fn_half1", "fn_half2"].includes(def.key) ||
  1834→            /^fn_(part|mic|mov)\d$/.test(def.key))
  1835→        ) {
  1836→          continue;
  1837→        }
  1838→        let dv: ParamValue;
  1839→        if (typeof def.default === "string") {
  1840→          // C++ placeholder defaults → ""
  1841→          dv = /^(std::|LABEL_|CURRENT_|NODE_)/.test(def.default) ? "" : def.default;
  1842→        } else {
  1843→          dv = def.default;
  1844→        }
  1845→        const radioOptions =
  1846→          def.type === "select" && def.radio && def.radio.length > 0 ? def.radio : undefined;
  1847→        const schemaType: ParamType =
  1848→          def.type === "select" && radioOptions
  1849→            ? "select"
  1850→            : def.type === "bool"
  1851→              ? "bool"
  1852→              : def.type === "number"
  1853→                ? "number"
  1854→                : def.type === "path"
  1855→                  ? "path"
  1856→                  : "text";
  1857→        if (schemaType === "select" && !radioOptions) continue;
  1858→        spec.params.push({
  1859→          key: def.key,
  1860→          label: def.label,
  1861→          type: schemaType,
  1862→          default: dv,
  1863→          min: def.min,
  1864→          max: def.max,
  1865→          step: def.step,
  1866→          options: radioOptions,
  1867→          tab: tab.label,
  1868→          advanced: o.expert,
  1869→          hint: relionHint(def.help),
  1870→        });
  1871→        have.add(def.key);
  1872→        populated = true;
  1873→      }
  1874→      if (populated && !relionTabOrder.includes(tab.label)) relionTabOrder.push(tab.label);
  1875→    }
  1876→    // tab order = RELION's own window order, with any curated-only tabs
  1877→    // appended where they already were (relative order preserved)
  1878→    const merged = [
  1879→      ...relionTabOrder,
  1880→      ...spec.tabs.filter((t) => !relionTabOrder.includes(t)),
  1881→    ];
  1882→    spec.tabs.length = 0;
  1883→    spec.tabs.push(...merged);
  1884→  }
  1885→}

  1943→/** Can the output port of one job type feed the input port of another? */
  1944→export function portsCompatible(
  1945→  fromType: string,
  1946→  fromPort: string,
  1947→  toType: string,
  1948→  toPort: string
  1949→): boolean {

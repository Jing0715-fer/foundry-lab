   147→/**
   148→ * Temporary "live wire" following the cursor while a connection is pending
   149→ * (click-click mode or drag-to-connect). Rendered inside the workspace so
   150→ * it scales with the zoom. Supports both wiring directions: "out" wires
   151→ * start at an output port (right edge), "in" wires start at an input port
   152→ * (left edge). The job-card pulse rings already signal the compatible
   153→ * ports on the other side. The SVG box hugs the anchor+cursor bbox with a
   154→ * viewBox that keeps workspace coordinates (infinite canvas).
   155→ */
   156→const LiveWire = React.memo(function LiveWire({
   157→  rootRef,
   158→  jobs,
   159→}: {
   160→  rootRef: React.RefObject<HTMLDivElement | null>;
   161→  jobs: JobDTO[];
   162→}) {
   163→  const pendingFrom = useWorkflowStore((s) => s.pendingFrom);
   164→  const [cursor, setCursor] = React.useState<{ x: number; y: number } | null>(null);
   165→
   166→  React.useEffect(() => {
   167→    setCursor(null);
   168→    const el = rootRef.current;
   169→    if (!el || !pendingFrom) return;
   170→    const onMove = (e: PointerEvent) => {
   171→      const rect = el.getBoundingClientRect();
   172→      const vp = useWorkflowStore.getState().viewport;
   173→      setCursor({
   174→        x: (e.clientX - rect.left - vp.x) / vp.zoom,
   175→        y: (e.clientY - rect.top - vp.y) / vp.zoom,
   176→      });
   177→    };
   178→    el.addEventListener("pointermove", onMove);
   179→    return () => el.removeEventListener("pointermove", onMove);
   180→  }, [pendingFrom, rootRef]);
   181→
   182→  if (!pendingFrom || !cursor) return null;
   183→  const job = jobs.find((j) => j.id === pendingFrom.jobId);
   184→  if (!job) return null;
   185→  const spec = jobType(job.type);
   186→
   187→  // "out" wires anchor at an output port (right edge); "in" wires anchor
   188→  // at an input port (left edge) and are dragged backwards to an output
   189→  let sx: number;
   190→  let sy: number;
   191→  if (pendingFrom.dir === "in") {
   192→    const inIdx = Math.max(0, spec?.inputs.findIndex((p) => p.name === pendingFrom.port) ?? 0);
   193→    const nIn = Math.max(1, spec?.inputs.length ?? 0);
   194→    sx = job.x;
   195→    sy = job.y + portY(inIdx, nIn);
   196→  } else {
   197→    const outIdx = Math.max(0, spec?.outputs.findIndex((p) => p.name === pendingFrom.port) ?? 0);
   198→    const nOut = Math.max(1, spec?.outputs.length ?? 0);
   199→    sx = job.x + CARD_W;
   200→    sy = job.y + portY(outIdx, nOut);
   201→  }
   202→
   203→  const bx = Math.min(sx, cursor.x) - WIRE_PAD;
   204→  const by = Math.min(sy, cursor.y) - WIRE_PAD;
   205→  const bw = Math.abs(cursor.x - sx) + 2 * WIRE_PAD;
   206→  const bh = Math.abs(cursor.y - sy) + 2 * WIRE_PAD;
   207→
   208→  return (
   209→    <svg
   210→      width={bw}
   211→      height={bh}
   212→      viewBox={`${bx} ${by} ${bw} ${bh}`}
   213→      className="pointer-events-none absolute left-0 top-0"
   214→      style={{ left: bx, top: by, overflow: "visible" }}
   215→      aria-hidden="true"
   216→    >
   217→      <circle cx={sx} cy={sy} r={4} fill="var(--primary)" opacity={0.9} />
   218→      <path
   219→        d={pendingWirePath(sx, sy, cursor.x, cursor.y, pendingFrom.dir === "in" ? "in" : "out")}
   220→        stroke="var(--primary)"
   221→        strokeWidth={2.5}
   222→        strokeLinecap="round"
   223→        strokeLinejoin="round"
   224→        opacity={0.8}
   225→        fill="none"
   226→        className="edge-flow"
   227→      />
   228→      <circle cx={cursor.x} cy={cursor.y} r={3} fill="var(--primary)" opacity={0.55} />
   229→    </svg>
   230→  );
   231→});

   256→/** Consecutive same-kind runs over one rendered row order. A pair of
   257→ *  moves is normal work — only HISTORY_GROUP_MIN+ collapses. */
   258→const HISTORY_GROUP_MIN = 3;
   259→
   260→function historyRuns(kinds: HistoryRowKind[], labels: string[]): HistoryRun[] {
   261→  const runs: HistoryRun[] = [];
   262→  for (let i = 0; i < kinds.length; i++) {
   263→    const last = runs[runs.length - 1];
   264→    if (last && last.kind === kinds[i]) {
   265→      last.end = i;
   266→      last.labels.push(labels[i]);
   267→    } else {
   268→      runs.push({ kind: kinds[i], start: i, end: i, labels: [labels[i]] });
   269→    }
   270→  }
   271→  return runs;
   272→}

   457→/**
   458→ * Task 129 — post-apply connection suggestions. An applied template
   459→ * lands as a wired island; this chip proposes the wires to its new
   460→ * neighbors (free boundary inputs × same-workspace free outputs),
   461→ * lists every pair EXPLICITLY, and wires only on Connect — a
   462→ * suggestion never connects anything by itself. Rows toggle inclusion
   463→ * (the excluded ones dim), the whole chip dismisses; navigation and
   464→ * vanished endpoints clear it silently.
   465→ */
   466→const TemplateSuggestionsChip = React.memo(function TemplateSuggestionsChip() {
   467→  const suggestions = useWorkflowStore((s) => s.templateSuggestions);
   468→  const activeWorkspaceId = useWorkflowStore((s) => s.activeWorkspaceId);
   469→  const jobs = useWorkflowStore((s) => s.jobs);
   470→  const [busy, setBusy] = React.useState(false);
   471→  const [excluded, setExcluded] = React.useState<Set<string>>(new Set());
   472→
   473→  // a new batch (different pairs) resets per-row inclusion — excluded
   474→  // rows belong to the batch they were excluded from
   475→  const batchKey = suggestions
   476→    ? suggestions.items.map((i) => `${i.fromJobId}>${i.toJobId}`).join(",")
   477→    : "";
   478→  React.useEffect(() => {
   479→    setExcluded(new Set());
   480→  }, [batchKey]);
   481→
   482→  // three silent exits: another workspace's batch, every endpoint already
   483→  // applied, or an endpoint vanished (job deleted) — a chip pointing at a
   484→  // ghost would be a lie the canvas renders
   485→  const alive = React.useMemo(() => {
   486→    if (!suggestions || suggestions.workspaceId !== (activeWorkspaceId ?? "")) return null;
   487→    const ids = new Set(jobs.map((j) => j.id));
   488→    const items = suggestions.items.filter((i) => ids.has(i.fromJobId) && ids.has(i.toJobId));
   489→    return items.length > 0 ? items : null;
   490→  }, [suggestions, activeWorkspaceId, jobs]);
   491→
   492→  if (!alive) return null;
   493→  const included = alive.filter((i) => !excluded.has(`${i.fromJobId}>${i.toJobId}`));
   494→  const connectIncluded = () => {
   495→    setBusy(true);
   496→    void useWorkflowStore
   497→      .getState()
   498→      .applyTemplateSuggestions(included)
   499→      .finally(() => setBusy(false));
   500→  };
   501→
   502→  return (
   503→    <div
   504→      data-canvas-ui="template-suggestions"
   505→      className="card-lift animate-rise absolute bottom-14 left-1/2 z-30 w-[min(420px,calc(100%-24px))] -translate-x-1/2 rounded-lg border bg-card/95 p-2 shadow-md backdrop-blur"
   506→    >
   507→      <div className="flex items-center gap-1.5">
   508→        <Waypoints className="size-3.5 shrink-0 text-primary" aria-hidden="true" />
   509→        <p className="flex-1 text-[11px] font-semibold text-muted-foreground">
   510→          {alive.length === 1 ? "1 suggested wire" : `${alive.length} suggested wires`}
   511→          <span className="ml-1.5 font-normal text-muted-foreground/70">
   512→            from the applied template — click a row to skip it
   513→          </span>
   514→        </p>
   515→        <button
   516→          type="button"
   517→          onClick={() => useWorkflowStore.getState().dismissTemplateSuggestions()}
   518→          aria-label="Dismiss connection suggestions"
   519→          title="Dismiss — wire by hand instead"
   520→          className="rounded-full p-0.5 text-muted-foreground transition-colors hover:text-foreground"
   521→          data-testid="template-suggestions-dismiss"

   905→  // canvas, mounting every card in ONE synchronous render pass blocked the
   906→  // main thread exactly while the user was trying to look at the app — the
   907→  // "opening the page stutters" report. Two levers, composed:
   908→  //   · useDeferredValue — the card layer re-renders at transition priority:
   909→  //     the shell paints first and pan/zoom/keys stay responsive while the
   910→  //     concurrent pass renders (steady-state polls stay free — the deferred
   911→  //     value only lags on real change, and memoized cards skip unchanged
   912→  //     rows anyway);
   913→  //   · chunked reveal — a landing that GROWS the list by more than a chunk
   914→  //     mounts cards ~20 per frame instead of one giant pop, so early cards
   915→  //     are visible while the rest stream in.
   916→  // Everything interactive (fit, print, export, find, LiveWire) keeps the
   917→  // FRESH lists; only the workspace render layer consumes these.
   918→  const deferredJobs = React.useDeferredValue(jobs);

   923→    // t364 — adjust state DURING render (React's sanctioned prev-state

   925→    // big list ever commits, so the first visible frame already shows the

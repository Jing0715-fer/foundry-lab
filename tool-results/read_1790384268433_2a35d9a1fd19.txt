    31→  envVars: Record<string, string>;

    35→function formatValue(v: string | number | boolean): string {
    36→  if (typeof v === "boolean") return v ? "true" : "false";
    37→  return String(v);
    38→}
    39→
    40→// Skip these meta-keys that are handled specially (not direct CLI flags)
    41→const SKIP_KEYS = new Set([
    42→  "protocol", "script_xml", "extra_options", "extra",
    43→]);
    44→
    45→export function buildCommand(
    46→  tool: ToolDefinition,
    47→  env: ToolEnvironment | null,
    48→  params: Record<string, string | number | boolean>
    49→): BuiltCommand {
    50→  const exePath = env?.exePath || tool.defaultExe;
    51→
    52→  switch (tool.cliStyle) {
    53→    // ============================================================
    54→    // Hydra style (RFdiffusion): group.key=value, lists need [ ]
    55→    // ============================================================
    56→    case "hydra": {
    57→      const parts: string[] = ["python", exePath];
    58→      const hasSymmetry = params["inference.symmetry"] && params["inference.symmetry"] !== "none";
    59→      if (hasSymmetry) parts.push("--config-name", "symmetry");
    60→
    61→      for (const f of tool.paramFields) {
    62→        if (SKIP_KEYS.has(f.key)) continue;
    63→        const v = params[f.key];
    64→        if (v === undefined || v === null || v === "" || v === "none") continue;
    65→        const flag = f.flag || f.key;
    66→
    67→        if (f.key === "inference.symmetry") {
    68→          // Already handled via --config-name; set the value
    69→          parts.push(`${flag}=${String(v)}`);
    70→          continue;
    71→        }
    72→
    73→        if (f.key === "inference.cyclic" && v === true) {
    74→          parts.push(`${flag}=True`);
    75→          continue;
    76→        }
    77→
    78→        // List-type params (contigs, inpaint_seq, hotspot_res, etc.) need [ ] wrapping
    79→        const isListParam = [
    80→          "contigmap.contigs", "contigmap.inpaint_seq", "contigmap.inpaint_str",
    81→          "contigmap.provide_seq", "ppi.hotspot_res", "potentials.guiding_potentials",
    82→        ].includes(f.key);
    83→
    84→        if (typeof v === "boolean") {
    85→          parts.push(`${flag}=${v ? "True" : "False"}`);
    86→        } else if (isListParam) {
    87→          // Wrap in single quotes + brackets for Hydra: 'key=[value]'
    88→          let val = String(v).trim();
    89→          if (!val.startsWith("[")) val = `[${val}]`;
    90→          parts.push(`'${flag}=${val}'`);
    91→        } else {
    92→          parts.push(`${flag}=${formatValue(v)}`);
    93→        }
    94→      }
    95→      return { command: parts.join(" "), envVars: {}, cwd: env?.workDir || undefined };
    96→    }
    97→
    98→    // ============================================================
    99→    // Click style (RFantibody): --flag value, --bool-flag for True
   100→    // ============================================================
   101→    case "click": {
   102→      const parts: string[] = [exePath];
   103→      for (const f of tool.paramFields) {
   104→        if (SKIP_KEYS.has(f.key)) continue;
   105→        const v = params[f.key];
   106→        if (v === undefined || v === null || v === "" || v === "none") continue;
   107→        const flag = f.flag || `--${f.key.replace(/_/g, "-")}`;
   108→
   109→        if (typeof v === "boolean") {
   110→          // Click bool flags: --deterministic (True), --no-trajectory (True)
   111→          if (v) parts.push(flag);
   112→        } else {
   113→          parts.push(flag, shellQuote(String(v)));
   114→        }
   115→      }
   116→      // Handle --extra (raw Hydra overrides, multiline → multiple -e flags)
   117→      const extraRaw = String(params["extra"] || "").trim();
   118→      if (extraRaw) {
   119→        for (const line of extraRaw.split("\n")) {
   120→          const trimmed = line.trim();
   121→          if (trimmed) parts.push("-e", shellQuote(trimmed));
   122→        }
   123→      }
   124→      return { command: parts.join(" "), envVars: {}, cwd: env?.workDir || undefined };
   125→    }
   126→
   127→    // ============================================================
   128→    // argparse style (ProteinMPNN): --flag value, --store_true for True
   129→    // ============================================================
   130→    case "argparse": {
   131→      const parts: string[] = ["python", exePath];
   132→      for (const f of tool.paramFields) {
   133→        const v = params[f.key];
   134→        if (v === undefined || v === null || v === "" || v === "none") continue;
   135→        const flag = f.flag || `--${f.key}`;
   136→
   137→        if (typeof v === "boolean") {
   138→          // argparse store_true: only include if True
   139→          if (v) parts.push(flag);
   140→        } else {
   141→          parts.push(flag, shellQuote(String(v)));
   142→        }
   143→      }
   144→      return { command: parts.join(" "), envVars: {}, cwd: env?.workDir || undefined };
   145→    }
   146→
   147→    // ============================================================
   148→    // Rosetta style: -flag value, -flag for bools, XML via temp file
   149→    // ============================================================
   150→    case "rosetta": {
   151→      const protocol = String(params["protocol"] || "relax");
   152→      const scriptXml = String(params["script_xml"] || "");
   153→      const extraOptions = String(params["extra_options"] || "").trim();
   154→      const parts: string[] = [exePath];
   155→
   156→      // XML protocol (temp file placeholder or explicit path)
   157→      if (protocol === "custom" && scriptXml) {
   158→        parts.push(`-parser:protocol`, shellQuote(scriptXml));
   159→      } else if (protocol !== "custom") {
   160→        parts.push(`-parser:protocol`, `__ROSETTA_XML__`);
   161→      }
   162→
   163→      // Script vars
   164→      const scriptVars = String(params["script_vars"] || "").trim();
   165→      if (scriptVars) parts.push(`-parser:script_vars`, shellQuote(scriptVars));
   166→
   167→      for (const f of tool.paramFields) {
   168→        if (SKIP_KEYS.has(f.key) || f.key === "script_vars") continue;
   169→        const v = params[f.key];
   170→        if (v === undefined || v === null || v === "" || v === "none") continue;
   171→        const flag = f.flag || f.key;
   172→        if (typeof v === "boolean") {
   173→          if (v) parts.push(`-${flag}`);
   174→        } else {
   175→          parts.push(`-${flag}`, shellQuote(String(v)));
   176→        }
   177→      }
   178→      if (extraOptions) parts.push(extraOptions);
   179→      return { command: parts.join(" "), envVars: {}, cwd: env?.workDir || undefined };
   180→    }
   181→
   182→    default:
   183→      return { command: exePath, envVars: {}, cwd: env?.workDir || undefined };
   184→  }
   185→}

   187→export function wrapForRuntime(
   188→  built: BuiltCommand,
   189→  env: ToolEnvironment | null
   190→): { fullCommand: string; cwd?: string } {
   191→  const runtime = env?.runtime || "native";
   192→  const condaEnv = env?.condaEnv;
   193→  const workDir = env?.workDir || built.cwd;
   194→
   195→  if (runtime === "native") {
   196→    return { fullCommand: built.command, cwd: workDir };
   197→  }
   198→
   199→  if (runtime === "conda") {
   200→    const lines: string[] = [];
   201→    if (condaEnv) lines.push(`conda activate ${shellQuote(condaEnv)}`);
   202→    if (workDir) lines.push(`cd ${shellQuote(workDir)}`);
   203→    lines.push(built.command);
   204→    return {
   205→      fullCommand: `bash -lc ${shellQuote(lines.join(" && "))}`,
   206→    };
   207→  }
   208→
   209→  // wsl
   210→  const lines: string[] = [];
   211→  if (condaEnv) lines.push(`conda activate ${shellQuote(condaEnv)}`);
   212→  if (workDir) lines.push(`cd ${shellQuote(wslifyPath(workDir))}`);
   213→  lines.push(built.command);
   214→  const distro = env?.wslDistro || "Ubuntu";
   215→  return {
   216→    fullCommand: `wsl -d ${distro} -- bash -lc ${shellQuote(lines.join(" && "))}`,
   217→  };
   218→}

   220→/**
   221→ * Resolve the WSL command into a spawnable { exe, args } tuple using only
   222→ * absolute paths. This avoids the Next.js 16 standalone worker's stripped PATH
   223→ * (which doesn't include C:\WINDOWS\system32), where `shell: true` and
   224→ * `shell: "cmd.exe"` both fail with ENOENT.
   225→ *
   226→ * Returns null for native/conda runtimes (use the regular shell path).
   227→ */
   228→export function resolveWslSpawn(
   229→  built: BuiltCommand,
   230→  env: ToolEnvironment | null
   231→): { exe: string; args: string[]; tailOutputFile?: string } | null {
   232→  if (!env || env.runtime !== "wsl") return null;
   233→
   234→  const condaEnv = env.condaEnv;
   235→  const workDir = env.workDir || built.cwd;
   236→  const distro = env.wslDistro || "Debian";
   237→
   238→  // Redirect the bash command's output to a Windows-side file. We then tail
   239→  // that file via setInterval polling (Next.js 16 standalone worker can't
   240→  // reliably stream stdout via child pipes for long-running commands).
   241→  const tailFile = path.join(
   242→    process.env.TEMP || process.env.TMP || "C:\\Users\\Public",
   243→    `vlab-job-${Date.now()}-${Math.random().toString(36).slice(2, 8)}.log`
   244→  );
   245→  // Use unix-style path inside WSL to write the file. The Windows path
   246→  // /mnt/<drive>/... is accessible from both sides.
   247→  const tailFileWsl = wslifyPath(tailFile);
   248→
   249→  const lines: string[] = [];
   250→  lines.push("PYTHONUNBUFFERED=1 PYTHONIOENCODING=utf-8");
   251→  if (condaEnv) lines.push(`conda activate ${shellQuote(condaEnv)}`);
   252→  if (workDir) lines.push(`cd ${shellQuote(wslifyPath(workDir))}`);
   253→  // exec redirects all output to file (>&1 merges stderr→stdout)
   254→  lines.push(`{ ${built.command}; } > ${shellQuote(tailFileWsl)} 2>&1`);
   255→
   256→  const wslExe = "C:/Windows/System32/wsl.exe";
   257→  const args = [
   258→    "-d",
   259→    distro,
   260→    "--",
   261→    "bash",
   262→    "-lc",
   263→    lines.join(" && "),
   264→  ];
   265→  return { exe: wslExe, args, tailOutputFile: tailFile };
   266→}
   267→
   268→function wslifyPath(p: string): string {
   269→  const m = /^([A-Za-z]):[\\/](.*)$/.exec(p);
   270→  if (m) return `/mnt/${m[1].toLowerCase()}/${m[2].replace(/\\/g, "/")}`;
   271→  return p;
   272→}
   273→
   274→function shellQuote(s: string): string {
   275→  if (/^[A-Za-z0-9_\-./:=+@,:]+$/.test(s)) return s;
   276→  return `'${s.replace(/'/g, `'"'"'`)}'`;
   277→}

   283→export interface RunHandle {
   284→  pid: number;
   285→  stdout: string;
   286→  stderr: string;
   287→  exitCode: number | null;
   288→}
   289→
   290→export interface RunOptions {
   291→  fullCommand: string;
   292→  cwd?: string;
   293→  extraEnv?: Record<string, string>;
   294→  onStdout?: (chunk: string) => void;
   295→  onStderr?: (chunk: string) => void;
   296→  onExit?: (code: number | null) => void;
   297→  rosettaXmlContent?: string | null;
   298→  timeoutMs?: number;
   299→  // Force line-buffered output (default true). Falls back to PYTHONUNBUFFERED
   300→  // + PYTHONIOENCODING injection for WSL wraps, and `python -u` for direct
   301→  // python invocations on the host. See runShell for the full logic.
   302→  forceLineBuffer?: boolean;
   303→  // Bypass shell entirely and spawn this exe with these args. Required when
   304→  // running inside Next.js 16 standalone, where the worker's PATH doesn't
   305→  // include C:\WINDOWS\system32 so `shell: true` / `shell: cmd.exe` fails
   306→  // with ENOENT. Caller is responsible for using absolute exe paths.
   307→  bypassShell?: { exe: string; args: string[] };
   308→  // When set together with bypassShell, the spawned command's stdout will be
   309→  // tailed from this file via setInterval polling instead of pipe events.
   310→  // Required for Next.js 16 standalone where child stdout streams stall on
   311→  // long-running processes.
   312→  tailOutputFile?: string;
   313→  // Use child_process.execFile (waits for close) instead of child_process.spawn
   314→  // (event-driven). execFile is reliable in Next 15/16 workers for short
   315→  // commands (used by detect-env successfully). For long-running tools, prefer
   316→  // tailOutputFile + spawn + exit-drain instead.
   317→  useExecFile?: boolean;
   318→}
   319→
   320→export function runShell(opts: RunOptions): {
   321→  pid: number;
   322→  done: Promise<RunHandle>;
   323→  kill: () => void;
   324→} {
   325→  let { fullCommand } = opts;
   326→
   327→  let xmlTempPath: string | null = null;
   328→  if (opts.rosettaXmlContent && fullCommand.includes("__ROSETTA_XML__")) {
   329→    xmlTempPath = path.join(opts.cwd || "/tmp", `rosetta_protocol_${Date.now()}.xml`);
   330→    fs.writeFile(xmlTempPath, opts.rosettaXmlContent).catch(() => {});
   331→    fullCommand = fullCommand.replace(/__ROSETTA_XML__/g, xmlTempPath);
   332→  }
   333→
   334→  // Wrap with stdbuf for line-buffered output. See forceLineBuffer docstring.
   335→  let effectiveCommand = fullCommand;
   336→  if (opts.forceLineBuffer !== false) {
   337→    if (fullCommand.includes("wsl ") || fullCommand.includes("wsl.exe")) {
   338→      effectiveCommand = fullCommand.replace(
   339→        /(bash -l[ci]+ ['"]?)/,
   340→        `$1PYTHONUNBUFFERED=1 PYTHONIOENCODING=utf-8 `
   341→      );
   342→    }
   343→    if (!fullCommand.includes("wsl ") && !fullCommand.includes("wsl.exe")) {
   344→      effectiveCommand = fullCommand.replace(/(^|\s)python\s+/g, "$1python -u ");
   345→    }
   346→  }
   347→
   348→  const env = {
   349→    ...process.env,
   350→    PYTHONUNBUFFERED: "1",
   351→    PYTHONIOENCODING: "utf-8",
   352→    ...opts.extraEnv,
   353→  };
   354→
   355→  // Decide how to spawn. When the caller provided bypassShell, use that
   356→  // directly (no shell resolver involved). Otherwise fall back to shell.
   357→  // When using bypassShell (WSL), do NOT set cwd to a Linux path —
   358→  // MSYS will mangle the spawn exe path. The WSL command itself
   359→  // already includes a `cd` to the correct working directory.
   360→  const spawnOptions: any = {
   361→    env,
   362→    stdio: ["ignore", "pipe", "pipe"],
   363→    windowsHide: true,
   364→  };
   365→  if (!opts.bypassShell && opts.cwd) {
   366→    spawnOptions.cwd = opts.cwd;
   367→  }
   368→
   369→  let child;
   370→  try {
   371→    if (opts.bypassShell) {
   372→      child = spawn(opts.bypassShell.exe, opts.bypassShell.args, spawnOptions);
   373→    } else {
   374→      spawnOptions.shell =
   375→        process.platform === "win32"
   376→          ? process.env.ComSpec || "C:/WINDOWS/system32/cmd.exe"
   377→          : "/bin/sh";
   378→      child = spawn(effectiveCommand, spawnOptions);
   379→    }
   380→  } catch (err: any) {
   381→    const msg = `[spawn failed] ${err?.message || err}\n`;
   382→    const failHandle: RunHandle = { pid: -1, stdout: "", stderr: msg, exitCode: -1 };
   383→    setTimeout(() => opts.onExit?.(-1), 0);
   384→    return { pid: -1, done: Promise.resolve(failHandle), kill: () => {} };
   385→  }
   386→
   387→  const handle: RunHandle = {
   388→    pid: child.pid || -1,
   389→    stdout: "",
   390→    stderr: "",
   391→    exitCode: null,
   392→  };
   393→
   394→  // ── Next.js 15 worker (Turbopack or webpack dev, plus standalone start)
   395→  //    cannot reliably receive long-running stdout via child pipe `data`
   396→  //    events — the events stall until the process completes, by which point
   397→  //    the user has already given up. Detect two cases:
   398→  //
   399→  //  A. `tailOutputFile` is provided (preferred for WSL/Conda commands):
   400→  //     the spawned command writes its stdout/stderr to a Windows-side
   401→  //     file. We poll the file with setInterval (best-effort) and ALWAYS
   402→  //     drain the full file on `child.on('exit')`. The exit-time drain is
   403→  //     the only reliable path inside Next's worker.
   404→  //
   405→  //  B. No tailOutputFile: fall back to child.stdout.on('data') events.
   406→  //     Works for short jobs; long jobs may stall until exit.
   407→  if (opts.bypassShell && opts.tailOutputFile) {
   408→    return runShellWithFileTail(opts, child, handle, env);
   409→  }
   410→
   411→  child.stdout?.on("data", (chunk: Buffer) => {
   412→    const s = chunk.toString();
   413→    handle.stdout += s;
   414→    opts.onStdout?.(s);
   415→  });
   416→  child.stderr?.on("data", (chunk: Buffer) => {
   417→    const s = chunk.toString();
   418→    handle.stderr += s;
   419→    opts.onStderr?.(s);
   420→  });
   421→  child.on("error", (err: any) => {
   422→    // Async spawn failure (e.g. ENOENT after spawn returned). Surface as a
   423→    // failed exit so the route persists the error and the UI sees it.
   424→    handle.stderr += `[spawn error] ${err?.message || err}\n`;
   425→    handle.exitCode = -1;
   426→    opts.onStderr?.(handle.stderr);
   427→    opts.onExit?.(-1);
   428→  });
   429→
   430→  let timedOut = false;
   431→  let timer: ReturnType<typeof setTimeout> | null = null;
   432→  if (opts.timeoutMs) {
   433→    timer = setTimeout(() => {
   434→      timedOut = true;
   435→      child.kill("SIGTERM");
   436→    }, opts.timeoutMs);
   437→  }
   438→
   439→  const done = new Promise<RunHandle>((resolve) => {
   440→    let settled = false;
   441→    const settle = (code: number | null, reason: string) => {
   442→      if (settled) return;
   443→      settled = true;
   444→      handle.exitCode = code ?? (reason ? -1 : null);
   445→      if (timedOut) handle.stderr += "\n[Process timed out and was terminated]\n";
   446→      if (reason && handle.exitCode !== 0) handle.stderr += `[${reason}]\n`;
   447→      if (xmlTempPath) fs.unlink(xmlTempPath).catch(() => {});
   448→      opts.onExit?.(handle.exitCode);
   449→      resolve(handle);
   450→    };
   451→    child.on("exit", (code, signal) => {
   452→      if (timer) clearTimeout(timer);
   453→      settle(code ?? (signal ? -1 : null), "exit");
   454→    });
   455→    child.on("error", (err: any) => {
   456→      if (timer) clearTimeout(timer);
   457→      handle.stderr += `[spawn error] ${err?.message || err}\n`;
   458→      settle(-1, "spawn error");
   459→    });
   460→  });
   461→
   462→  const kill = () => {
   463→    try {
   464→      child.kill("SIGTERM");
   465→    } catch {
   466→      /* ignore */
   467→    }
   468→  };
   469→
   470→  return { pid: handle.pid, done, kill };
   471→}
   472→
   473→/**
   474→ * File-tail variant for runs where stdout streaming via child pipes is unreliable
   475→ * (Next.js 16 standalone worker / Turbopack dev). The command runs and writes
   476→ * stdout/stderr to `tailOutputFile`; we read the file in three places:
   477→ *   1. On a setInterval poll (best-effort streaming)
   478→ *   2. Lazily on each `child.stdout.on('data')` event (fallback if pipe events fire)
   479→ *   3. On `child.on('exit')` we drain the full file before signaling completion
   480→ * The interval is best-effort; if Next.js 16's worker event loop doesn't run
   481→ * timers reliably, we still get the full output at exit time.
   482→ */
   483→function runShellWithFileTail(
   484→  opts: RunOptions,
   485→  child: ReturnType<typeof spawn>,
   486→  handle: RunHandle,
   487→  env: NodeJS.ProcessEnv
   488→): { pid: number; done: Promise<RunHandle>; kill: () => void } {
   489→  const outFile = opts.tailOutputFile!;
   490→  let lastOutSize = 0;
   491→
   492→  const readNewBytes = async () => {
   493→    try {
   494→      const stat = await fs.stat(outFile);
   495→      if (stat.size > lastOutSize) {
   496→        const fd = await fs.open(outFile, "r");
   497→        try {
   498→          const buf = Buffer.alloc(stat.size - lastOutSize);
   499→          await fd.read(buf, 0, buf.length, lastOutSize);
   500→          const s = buf.toString("utf8");
   501→          handle.stdout += s;
   502→          opts.onStdout?.(s);
   503→          lastOutSize = stat.size;
   504→        } finally {
   505→          await fd.close();
   506→        }
   507→      }
   508→    } catch { /* file may not exist yet */ }
   509→  };
   510→
   511→  // Best-effort streaming poll. Next.js 16 worker may not run this reliably;
   512→  // we treat any failure as non-fatal and rely on exit-time drain below.
   513→  let interval: ReturnType<typeof setInterval> | null = null;
   514→  try {
   515→    interval = setInterval(() => { readNewBytes().catch(() => {}); }, 500);
   516→  } catch { /* ignore */ }
   517→
   518→  let timedOut = false;
   519→  let timer: ReturnType<typeof setTimeout> | null = null;
   520→  if (opts.timeoutMs) {
   521→    timer = setTimeout(() => {
   522→      timedOut = true;
   523→      child.kill("SIGTERM");
   524→    }, opts.timeoutMs);
   525→  }
   526→
   527→  const done = new Promise<RunHandle>((resolve) => {
   528→    child.on("exit", async (code, signal) => {
   529→      if (interval) clearInterval(interval);
   530→      if (timer) clearTimeout(timer);
   531→      // CRITICAL: drain the full file at exit time, regardless of whether the
   532→      // interval poll was running. This is the reliable path inside Next 16
   533→      // workers where pipe events stall and timers may not fire often enough.
   534→      await readNewBytes();
   535→      handle.exitCode = code ?? (signal ? -1 : null);
   536→      if (timedOut) {
   537→        handle.stderr += "\n[Process timed out and was terminated]\n";
   538→      }
   539→      opts.onExit?.(handle.exitCode);
   540→      resolve(handle);
   541→    });
   542→  });
   543→
   544→  return {
   545→    pid: handle.pid,
   546→    done,
   547→    kill: () => {
   548→      if (interval) clearInterval(interval);
   549→      try { child.kill("SIGTERM"); } catch {}
   550→    },
   551→  };
   552→}

   554→/**
   555→ * Collect output files from a directory. If the directory is a WSL-internal
   556→ * path (doesn't exist on the Windows host, e.g. /tmp/... or /root/... or
   557→ * /home/<user>/... but NOT /mnt/c/... which is the Windows mount), we shell
   558→ * out to `wsl` to list and stat the files.
   559→ *
   560→ * This matters because tools like RFdiffusion default to /tmp/... and Rosetta
   561→ * to its build directory. The UI tries to surface these via the "Recent
   562→ * Outputs" file picker, so they need to be discoverable from Windows.
   563→ */
   564→export async function collectOutputFiles(outputDir: string): Promise<{ path: string; size: number }[]> {
   565→  const isWslInternal =
   566→    outputDir.startsWith("/tmp/") ||
   567→    outputDir.startsWith("/var/") ||
   568→    outputDir.startsWith("/root/") ||
   569→    outputDir.startsWith("/home/") ||
   570→    outputDir.startsWith("/opt/");
   571→
   572→  if (isWslInternal) {
   573→    return collectOutputFilesViaWsl(outputDir);
   574→  }
   575→
   576→  // Native or /mnt/... path — use host fs
   577→  try {
   578→    const entries = await fs.readdir(outputDir, { withFileTypes: true });
   579→    const files: { path: string; size: number }[] = [];
   580→    for (const e of entries) {
   581→      if (e.isFile()) {
   582→        const full = path.join(outputDir, e.name);
   583→        const stat = await fs.stat(full);
   584→        if (/\.(pdb|sc|json|fa|fasta|cif|xml|txt|log)$/i.test(e.name)) {
   585→          files.push({ path: full, size: stat.size });
   586→        }
   587→      }
   588→    }
   589→    return files;
   590→  } catch {
   591→    return [];
   592→  }
   593→}
   594→
   595→async function collectOutputFilesViaWsl(outputDir: string): Promise<{ path: string; size: number }[]> {
   596→  // Use a single wsl invocation that lists files + sizes (avoid one exec per file).
   597→  // `find` with `-printf` is portable across GNU find (default on Debian/Ubuntu WSL).
   598→  const escaped = outputDir.replace(/'/g, `'\\''`);
   599→  const cmd = `wsl -d Debian -- bash -lc "find '${escaped}' -maxdepth 1 -type f -printf '%p\\\\t%s\\\\n' 2>/dev/null"`;
   600→  try {
   601→    const { spawn } = await import("child_process");
   602→    const result = await new Promise<{ stdout: string; stderr: string }>((resolve, reject) => {
   603→      const p = spawn(cmd, { shell: true });
   604→      let stdout = "";
   605→      let stderr = "";
   606→      p.stdout?.on("data", (c: Buffer) => (stdout += c.toString()));
   607→      p.stderr?.on("data", (c: Buffer) => (stderr += c.toString()));
   608→      p.on("exit", (code) => (code === 0 ? resolve({ stdout, stderr }) : reject(new Error(`wsl find exit ${code}: ${stderr}`))));
   609→      p.on("error", reject);
   610→      setTimeout(() => p.kill(), 15000);
   611→    });
   612→    const files: { path: string; size: number }[] = [];
   613→    for (const line of result.stdout.split(/\r?\n/)) {
   614→      const m = line.match(/^(.+)\t(\d+)$/);
   615→      if (m && /\.(pdb|sc|json|fa|fasta|cif|xml|txt|log)$/i.test(m[1])) {
   616→        files.push({ path: m[1], size: parseInt(m[2], 10) });
   617→      }
   618→    }
   619→    return files;
   620→  } catch {
   621→    return [];
   622→  }
   623→}

   709→export function extractToolCalls(text: string): ToolCallRequest[] {
   710→  const calls: ToolCallRequest[] = [];
   711→  const re = /```tool\s*\n([\s\S]*?)```/g;
   712→  let m: RegExpExecArray | null;
   713→  while ((m = re.exec(text)) !== null) {
   714→    try {
   715→      const parsed = JSON.parse(m[1].trim());
   716→      if (parsed && typeof parsed === "object" && "tool" in parsed) {
   717→        calls.push(parsed as ToolCallRequest);
   718→      }
   719→    } catch {
   720→      /* ignore malformed */
   721→    }
   722→  }
   723→  return calls;
   724→}

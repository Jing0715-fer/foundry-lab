   152→export function RemoteRunButton({
   153→  job,
   154→  dialogOnly = false,
   155→  open: openProp,
   156→  onOpenChange,
   157→}: {
   158→  job: JobDTO;
   159→  /** render just the dialog (no trigger) — for controlled callers. */
   160→  dialogOnly?: boolean;
   161→  /** controlled open state (requires dialogOnly). */
   162→  open?: boolean;
   163→  onOpenChange?: (open: boolean) => void;
   164→}) {
   165→  const [openState, setOpenState] = React.useState(false);
   166→  const open = dialogOnly ? (openProp ?? false) : openState;
   167→  const setOpen = (v: boolean) => {
   168→    setOpenState(v);
   169→    onOpenChange?.(v);
   170→  };
   171→  const { connections, reload } = useRemoteConnections(open);
   172→  const runJobRemote = useWorkflowStore((s) => s.runJobRemote);
   173→  // t300 — a REMOTE project's bound cluster is the dialog's DEFAULT (and
   174→  // locked) target: the project's picked input paths are absolute on THAT
   175→  // cluster — another connection would strand the data.
   176→  const projectRemote = useWorkflowStore((s) => s.project?.remote ?? null);
   177→
   178→  const [connId, setConnId] = React.useState("");
   179→  const [module, setModule] = React.useState("");

   182→  const [gpus, setGpus] = React.useState(6);
   183→  // t300 — the detected node group (Slurm partition) this sbatch pins;
   184→  // "__auto__" = the scheduler picks (falls back to the connection default)
   185→  const [partition, setPartition] = React.useState<string>(PARTITION_AUTO);
   186→  // t332 — the node picked straight from the live usage list (the panel's
   187→  // rows): an explicit --nodelist that survives multi-host groups. The
   188→  // full usage row is kept (its gpuTotal caps the stepper); null = no pin
   189→  // (the t300 single-host derivation applies).
   190→  const [pickedNode, setPickedNode] = React.useState<SlurmNodeUsage | null>(null);
   191→  // t306 — the array split (1 = off — the single-job contract; 2..64 shards
   192→  // ride --array=1-N%4). Defaults OFF: a re-run of an old job must submit
   193→  // byte-identical scripts unless the user asks for the split.
   194→  const [shards, setShards] = React.useState(1);
   195→  const [customModule, setCustomModule] = React.useState("");
   196→  const [pending, setPending] = React.useState(false);
   197→  // the nested cluster manager (empty state → add a connection right here)
   198→  const [clusterOpen, setClusterOpen] = React.useState(false);
   199→
   200→  const conn = connections.find((c) => c.id === connId) ?? null;
   201→  const probedModules = conn?.lastProbe?.relionModules ?? [];
   202→  const slurmAvailable = conn?.lastProbe?.slurm === true;
   203→  const partitionInventory = conn?.lastProbe?.slurmGpus ?? [];
   204→
   205→  // default connection: the project's binding (locked, see below), else
   206→  // keep the current pick, else the ACTIVE one, else first
   207→  React.useEffect(() => {
   208→    if (!open) return;
   209→    if (projectRemote) {
   210→      // the project's cluster wins unconditionally while it exists in the
   211→      // registry — a remote project's data lives THERE
   212→      if (connections.some((c) => c.id === projectRemote.connectionId)) {
   213→        setConnId(projectRemote.connectionId);
   214→        return;
   215→      }
   216→    }
   217→    setConnId((prev) => {
   218→      if (prev && connections.some((c) => c.id === prev)) return prev;
   219→      const active = readActiveRemoteConnectionId();
   220→      if (active && connections.some((c) => c.id === active)) return active;
   221→      return connections[0]?.id ?? "";
   222→    });
   223→  }, [open, connections, projectRemote]);
   224→
   225→  // module default follows the connection (its defaultModule, else first probed)
   226→  React.useEffect(() => {
   227→    if (!conn) {
   228→      setModule("");
   229→      return;
   230→    }
   231→    const probed = conn.lastProbe?.relionModules ?? [];
   232→    if (conn.defaultModule && probed.includes(conn.defaultModule)) setModule(conn.defaultModule);
   233→    else setModule(probed[0] ?? conn.defaultModule ?? "");
   234→  }, [conn]);
   235→
   236→  // t297 — the mode defaults to the connection's preference, and degrades
   237→  // to direct when this cluster offers no Slurm client (the probe's word).
   238→  React.useEffect(() => {
   239→    if (!conn) return;
   240→    if (conn.useSlurm && conn.lastProbe?.slurm) setMode("slurm");
   241→    else setMode("direct");
   242→  }, [conn]);
   243→
   244→  // t300 — partition default: the connection's pinned partition when the
   245→  // probe's inventory actually lists it, else auto
   246→  React.useEffect(() => {
   247→    if (!conn) return;
   248→    setPartition(
   249→      conn.slurmPartition && partitionInventory.some((p) => p.partition === conn.slurmPartition)
   250→        ? conn.slurmPartition
   251→        : PARTITION_AUTO
   252→    );
   253→  }, [conn, partitionInventory]);
   254→
   255→  // t332 — a node pin belongs to ONE cluster's inventory: a different
   256→  // connection's nodes are a different world, and the pin must not
   257→  // survive the switch (the panel's rows re-fetch underneath it)
   258→  React.useEffect(() => {
   259→    setPickedNode(null);
   260→  }, [conn?.id]);
   261→
   262→  // t300 — the GPU width's ceiling: the SELECTED node group's per-node
   263→  // GPUs (sinfo), falling back to the widest partition the probe saw, else 8.
   264→  // t332 — a node PINNED from the usage list caps the stepper with THAT
   265→  // node's own GPUs (scontrol's word — a 6-GPU node cannot honor
   266→  // --gres=gpu:8 however wide its partition's other machines are; a
   267→  // 0-GPU node keeps the old ceiling and lets the ask line name the
   268→  // contradiction instead).
   269→  const selectedGroup =
   270→    partition === PARTITION_AUTO ? null : partitionInventory.find((p) => p.partition === partition) ?? null;
   271→  const nodePin = pickedNode?.node ?? null;
   272→  // t337 — AUTO is not "the inventory's first group": with no picked
   273→  // partition (and no pin) the script still carries the CONNECTION'S
   274→  // default (conn.slurmPartition — exactly what the engine's
   275→  // effectivePartition resolves to). The stepper's ceiling must be the
   276→  // group the sbatch will actually name, or it offers widths the
   277→  // controller refuses at submit time ("auto" + 6 on a 5-GPU default
   278→  // partition was the user's live receipt).
   279→  const autoGroup = conn?.slurmPartition
   280→    ? partitionInventory.find((p) => p.partition === conn?.slurmPartition) ?? null
   281→    : null;
   282→  const maxGpus = Math.max(
   283→    1,
   284→    Math.min(
   285→      8,
   286→      pickedNode && pickedNode.gpuTotal > 0
   287→        ? pickedNode.gpuTotal
   288→        : (selectedGroup?.gpusPerNode ?? autoGroup?.gpusPerNode ?? partitionInventory[0]?.gpusPerNode ?? 8)
   289→    )
   290→  );
   291→  // clamping the pick when the partition changes (a 5-GPU partition cannot
   292→  // honor a 6-GPU request — the stepper must not offer it)
   293→  React.useEffect(() => {
   294→    setGpus((g) => Math.min(g, maxGpus));
   295→  }, [maxGpus]);
   296→
   297→  const disabled = job.status === "running" || job.linkedJobId != null;
   298→
   299→  // t306 — the array stepper only exists for eligible types (the dispatch
   300→  // would refuse the split for anything else) and only in slurm mode.
   301→  const arrayEligible = ARRAY_ELIGIBLE_TYPES.has(job.type);
   302→
   303→  // t320 — a LoG Auto-picking job is CPU-only (RELION's autopicker.cpp
   304→  // refuses --gpu on the Laplacian-of-Gaussian picker outright). The GPU
   305→  // stepper would be a knob the dispatch refuses at submit time — "a knob
   306→  // that would be refused is not a knob, it is a trap" (the array row's
   307→  // own doctrine): show the CPU contract instead. References/Topaz
   308→  // picking keep the stepper.
   309→  const logPick = isLogAutopick(job.type, job.params);
   310→
   311→  /** The module the dispatch will actually load: the free-text door wins
   312→   *  over the select while it carries a value. */
   313→  const effectiveModule =
   314→    module === CUSTOM_MODULE_VALUE ? customModule.trim() : module;
   315→
   316→  // t326 — the WIDTH truth: the dispatch honors the GPU-stepper width only
   317→  // for the MPI-multi-GPU types (relion_refine_mpi splits particles across
   318→  // ranks); everything else sizes itself (one GPU per task for motioncorr
   319→  // and References-picking, CPU tasks for ctffind/extract/LoG, single-GPU
   320→  // for the Topaz-style trainers). A stepper promising a width the sbatch
   321→  // would not write is a trap, not a knob — t320's own doctrine, applied
   322→  // to the width. The table is the SAME pure module the strategy derives
   323→  // from (gpu-width.ts): the dialog cannot drift from the dispatch.
   324→  const widthTruth = slurmWidthFor(job.type, { gpus, logAutopick: logPick });
   325→  // and the width is only real when the chosen MODULE can actually run
   326→  // MPI (a no-mpirun module falls back to a sequential single-GPU run —
   327→  // the probe's relionMpi map says which modules carry mpirun)
   328→  const moduleHasMpi = !!conn?.lastProbe?.relionMpi?.[effectiveModule];
   329→  const widthIsReal = !logPick && widthTruth.mode === "multi-gpu" && moduleHasMpi;
   330→  // t349 — the MPI rank count the dispatch writes for a real width:
   331→  // N+1 for N ≥ 2 (RELION's dedicated-master layout — rank 0 is the CPU
   332→  // master, one worker per card), 1 for width 1 (a single process; no
   333→  // split to make on one card). The preview speaks this number, the
   334→  // sbatch carries it (CF_RANKS in the script).
   335→  const mpiRankCount = gpus >= 2 ? gpus + 1 : 1;
   336→
   337→  // t326 — the submission preview's partition: the picked group, else the
   338→  // connection's pinned default (exactly what the dispatch's
   339→  // effectivePartition resolves to — the preview never invents a flag).
   340→  // t332 — an explicit node pin speaks for itself: the connection's
   341→  // default partition must not ride along (the engine suppresses it the
   342→  // same way — a --partition=normal + --nodelist=brain3 combo is refused
   343→  // at submit time; the node's own partition is where it lands).
   344→  // t340 — while a usage-list pin speaks, the node's OWN home is the
   345→  // partition that rides — the partition STATE (possibly the
   346→  // auto-initialized connection default) must not compose a contradiction
   347→  // the preview would then lie about. Unknown home → no partition
   348→  // line: the default decides, honestly shown.
   349→  const previewPartition = nodePin
   350→    ? (pickedNode?.partitions?.[0] ?? null)
   351→    : partition !== PARTITION_AUTO
   352→      ? partition
   353→      : (conn?.slurmPartition ?? null);
   354→
   355→  // t326 — the sbatch directives these knobs produce. Every entry here is
   356→  // one the dispatch's script builder actually writes (remote-run.ts):
   357→  // --partition (picked or connection default), --nodelist (single-host
   358→  // group), --gres/--ntasks (the MPI width), --array (the split). The
   359→  // preview is a promise the sbatch keeps.
   360→  const sbatchDirectives: string[] = [];
   361→  if (mode === "slurm") {
   362→    if (previewPartition) sbatchDirectives.push(`--partition=${previewPartition}`);
   363→    // t332 — the explicit node pick (the usage list's rows) pins the node
   364→    // outright; the t300 single-host group derivation stands only without
   365→    // one — the user's later, more specific choice speaks
   366→    if (nodePin) sbatchDirectives.push(`--nodelist=${nodePin}`);
   367→    else if (selectedGroup && selectedGroup.hosts?.length === 1)
   368→      sbatchDirectives.push(`--nodelist=${selectedGroup.hosts[0]}`);
   369→    if (logPick) sbatchDirectives.push("--ntasks=1");
   370→    else if (widthIsReal) {
   371→      // t349 — the dedicated-master layout: N workers + 1 CPU master
   372→      sbatchDirectives.push(`--ntasks=${mpiRankCount}`, `--gres=gpu:${gpus}`);
   373→    } else if (widthTruth.gpus === 1) {
   374→      // single-GPU steps, array-1-GPU shards, and the no-mpirun fallback
   375→      // all land the same shape: one GPU task
   376→      sbatchDirectives.push("--ntasks=1", "--gres=gpu:1");
   377→    } else {
   378→      sbatchDirectives.push("--ntasks=1");
   379→    }
   380→    if (arrayEligible && shards >= 2) sbatchDirectives.push(`--array=1-${shards}%4`);
   381→  }
   382→
   383→  // t327 — what THIS submission needs from the cluster right now, for
   384→  // the live usage panel's ask line: per-task GPUs (the width truth —
   385→  // the stepper's pick where real, the honest width elsewhere) × the
   386→  // simultaneous tasks (an array's shards capped by the %4 concurrency
   387→  // the sbatch actually writes).
   388→  const usageAsk = React.useMemo(() => {
   389→    const perTask = logPick ? 0 : widthIsReal ? gpus : widthTruth.gpus;
   390→    const tasks = arrayEligible && shards >= 2 ? Math.min(shards, 4) : 1;
   391→    return { gpus: perTask, tasks };
   392→  }, [logPick, widthIsReal, gpus, widthTruth.gpus, arrayEligible, shards]);
   393→
   394→  // t326 — radiogroup keyboard grammar: arrows move between the two mode
   395→  // cards (right/down → slurm when the cluster has a client, left/up →
   396→  // direct). Click/Enter/Space ride the native button semantics.
   397→  const onModeCardKey = (e: React.KeyboardEvent<HTMLElement>) => {
   398→    if (e.key === "ArrowRight" || e.key === "ArrowDown") {
   399→      e.preventDefault();
   400→      if (slurmAvailable) setMode("slurm");
   401→    } else if (e.key === "ArrowLeft" || e.key === "ArrowUp") {
   402→      e.preventDefault();
   403→      setMode("direct");
   404→    }
   405→  };
   406→
   407→  // t326 — the honest width box: what the dispatch will actually request
   408→  // when the stepper's width is not real for this type. Same shape as the
   409→  // LoG CPU box (t320): the contract, stated — not a knob that lies.
   410→  const widthBox = (() => {
   411→    if (widthTruth.mode === "multi-gpu" && !moduleHasMpi)
   412→      return {
   413→        headline: "1 × GPU — this module has no mpirun",
   414→        detail:
   415→          `relion_refine_mpi needs the module's own mpirun — ${effectiveModule || "this module"} ` +
   416→          "was not probed with one, so the dispatch runs the sequential single-GPU path " +
   417→          "(--gres=gpu:1, --ntasks=1, no mpirun). Pick an MPI-capable module to unlock the width stepper.",
   418→      };
   419→    if (widthTruth.mode === "single")
   420→      return {
   421→        headline: "1 × GPU — single-GPU step",
   422→        detail:
   423→          "This step runs on exactly one GPU (--gres=gpu:1, --ntasks=1) — " +
   424→          (job.type === "topaztrain"
   425→            ? "Topaz training converges on one GPU; multi-GPU would need DDP."
   426→            : "the external model works one GPU per job."),
   427→      };
   428→    if (widthTruth.mode === "array" && widthTruth.gpus === 1)
   429→      return {
   430→        headline: "1 × GPU per task",
   431→        detail:
   432→          "Each array shard runs on one GPU (--ntasks=1, --gres=gpu:1, no mpirun) — " +
   433→          "the real parallelism knob is the Array split below (more shards, not wider jobs).",
   434→      };
   435→    // t338 — the extract-specific CPU contract. The field question:
   436→    // 「extraction无法用GPU吗？」 — no: relion_preprocess (RELION's
   437→    // extraction engine) has NO GPU code path; box cutting + normalization
   438→    // run on the CPU, so a GPU allocation would just sit idle (and shrink
   439→    // the pool the classification jobs need). The honest speed knob is
   440→    // the Array split: N CPU shards, each extracting its share of the
   441→    // micrographs in parallel.
   442→    if (job.type === "extract")
   443→      return {
   444→        headline: "0 × GPU — CPU-only extraction",
   445→        detail:

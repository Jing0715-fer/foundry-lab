   110→/** Debounced PATCH for node updates. */
   111→function useDebouncedPatch() {
   112→  const timer = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   113→  return React.useCallback((id: string, body: Record<string, unknown>, delay = 350) => {
   114→    if (timer.current) clearTimeout(timer.current);
   115→    timer.current = setTimeout(async () => {
   116→      try {
   117→        await fetch(`/api/workflow/nodes/${id}`, {
   118→          method: "PATCH",
   119→          headers: { "Content-Type": "application/json" },
   120→          body: JSON.stringify(body),
   121→        });
   122→      } catch {
   123→        /* swallow — optimistic state already applied */
   124→      }
   125→    }, delay);
   126→  }, []);
   127→}

   163→/** Render one ParamSchema field. */
   164→function ParamField({
   165→  param,
   166→  value,
   167→  onChange,
   168→  agents,
   169→}: {
   170→  param: ParamSchema;
   171→  value: string | number | boolean | undefined;
   172→  onChange: (v: string | number | boolean) => void;
   173→  agents: AgentDTO[];
   174→}) {
   175→  const v = value ?? param.default;
   176→
   177→  switch (param.type) {
   178→    case "number":
   179→      return (
   180→        <Input
   181→          type="number"
   182→          value={typeof v === "number" ? v : Number(v) || 0}
   183→          min={param.min}
   184→          max={param.max}
   185→          step={param.step}
   186→          onChange={(e) => onChange(Number(e.target.value))}
   187→          className="h-8 text-sm"
   188→        />
   189→      );
   190→    case "select":
   191→      return (
   192→        <Select value={String(v ?? "")} onValueChange={onChange}>
   193→          <SelectTrigger className="w-full" size="sm">
   194→            <SelectValue placeholder="Select…" />
   195→          </SelectTrigger>
   196→          <SelectContent>
   197→            {(param.options ?? []).map((opt) => (
   198→              <SelectItem key={opt} value={opt}>
   199→                {opt}
   200→              </SelectItem>
   201→            ))}
   202→          </SelectContent>
   203→        </Select>
   204→      );
   205→    case "bool":
   206→      return (
   207→        <div className="flex h-8 items-center">
   208→          <Switch checked={Boolean(v)} onCheckedChange={onChange} />
   209→          <span className="ml-2 text-xs text-muted-foreground">
   210→            {Boolean(v) ? "on" : "off"}
   211→          </span>
   212→        </div>
   213→      );
   214→    case "textarea":
   215→      return (
   216→        <Textarea
   217→          value={String(v ?? "")}
   218→          onChange={(e) => onChange(e.target.value)}
   219→          placeholder={param.placeholder}
   220→          className="min-h-20 text-sm"
   221→        />
   222→      );
   223→    case "text":
   224→    case "path":
   225→      return (
   226→        <Input
   227→          type="text"
   228→          value={String(v ?? "")}
   229→          onChange={(e) => onChange(e.target.value)}
   230→          placeholder={param.placeholder}
   231→          className="h-8 text-sm"
   232→        />
   233→      );
   234→    default:
   235→      return null;
   236→  }
   237→}

   239→/** A single labeled param row. */
   240→function ParamRow({
   241→  param,
   242→  node,
   243→  agents,
   244→  onPatchParam,
   245→}: {
   246→  param: ParamSchema;
   247→  node: NodeDTO;
   248→  agents: AgentDTO[];
   249→  onPatchParam: (key: string, value: string | number | boolean) => void;
   250→}) {
   251→  // For agent nodes, render the refId param as an agent picker.
   252→  if (node.type === "agent" && param.key === "refId") {
   253→    return (
   254→      <div className="grid gap-1.5">
   255→        <label className="text-xs font-medium">{param.label}</label>
   256→        <AgentPicker
   257→          agents={agents}
   258→          value={String(node.params.refId ?? "")}
   259→          onChange={(v) => onPatchParam("refId", v)}
   260→        />
   261→        {param.hint && <p className="text-[11px] text-muted-foreground">{param.hint}</p>}
   262→      </div>
   263→    );
   264→  }
   265→
   266→  // For comptool nodes, hide param fields whose prefix doesn't match the current toolKey.
   267→  if (node.type === "comptool" && param.key.startsWith("param_")) {
   268→    const toolKey = String(node.params.toolKey ?? "rfdiffusion");
   269→    const prefix = `param_${toolKey}_`;
   270→    if (!param.key.startsWith(prefix)) return null;
   271→  }
   272→
   273→  return (
   274→    <div className="grid gap-1.5">
   275→      <label className="flex items-center justify-between text-xs font-medium">
   276→        <span>{param.label}</span>
   277→        {param.unit && (
   278→          <span className="text-[10px] font-normal text-muted-foreground">
   279→            {param.unit}
   280→          </span>
   281→        )}
   282→      </label>
   283→      <ParamField
   284→        param={param}
   285→        value={node.params[param.key]}
   286→        agents={agents}
   287→        onChange={(v) => onPatchParam(param.key, v)}
   288→      />
   289→      {param.hint && <p className="text-[11px] text-muted-foreground">{param.hint}</p>}
   290→    </div>
   291→  );
   292→}

   360→/** Logs tab content. Renders live-updating logs as they stream in via SSE
   361→ * (the running case used to show only a "Running…" spinner — now we surface
   362→ * whatever logs the backend has emitted so far, with a small live indicator
   363→ * when the buffer is still empty).
   364→ *
   365→ * The <pre> is auto-scrolled to the bottom on each update so the user sees
   366→ * the latest log line without manually scrolling. A "Download" button
   367→ * exports the current logs buffer as a .txt file (handy for sharing run
   368→ * output in bug reports). */
   369→function LogsTab({ node }: { node: NodeDTO }) {
   370→  const logs = node.logs?.trim();
   371→  const preRef = React.useRef<HTMLPreElement | null>(null);
   372→
   373→  // Auto-scroll to the bottom on each log update — but only if the user is
   374→  // already near the bottom (within 60px). This preserves scroll position
   375→  // when the user is reading older log lines.
   376→  React.useEffect(() => {
   377→    const el = preRef.current;
   378→    if (!el) return;
   379→    const nearBottom =
   380→      el.scrollHeight - el.scrollTop - el.clientHeight < 60;
   381→    if (nearBottom) {
   382→      el.scrollTop = el.scrollHeight;
   383→    }
   384→  }, [logs]);
   385→
   386→  const handleDownload = () => {
   387→    if (!logs) return;
   388→    const blob = new Blob([logs], { type: "text/plain;charset=utf-8" });
   389→    const url = URL.createObjectURL(blob);
   390→    const a = document.createElement("a");
   391→    a.href = url;
   392→    a.download = `${node.name.replace(/[^\w-]+/g, "_")}_logs.txt`;
   393→    document.body.appendChild(a);
   394→    a.click();
   395→    document.body.removeChild(a);
   396→    URL.revokeObjectURL(url);
   397→  };
   398→
   399→  if (!logs && node.status === "running") {
   400→    return (
   401→      <div className="flex items-center gap-2 p-4 text-xs text-muted-foreground">
   402→        <Loader2 className="size-3.5 animate-spin" />
   403→        Starting…
   404→      </div>
   405→    );
   406→  }
   407→  if (!logs) {
   408→    return (
   409→      <div className="p-4 text-center text-xs text-muted-foreground">
   410→        No logs yet.
   411→      </div>
   412→    );
   413→  }
   414→  return (
   415→    <div className="flex flex-col gap-2 p-3">
   416→      <div className="flex items-center justify-between">
   417→        <p className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   418→          logs · {logs.length.toLocaleString()} chars
   419→          {node.status === "running" && (
   420→            <span className="ml-2 inline-flex items-center gap-1 text-amber-600 dark:text-amber-400">
   421→              <Loader2 className="size-2.5 animate-spin" /> live
   422→            </span>
   423→          )}
   424→        </p>
   425→        <Button
   426→          variant="ghost"
   427→          size="sm"
   428→          className="h-7 gap-1 px-2 text-[11px]"
   429→          onClick={handleDownload}
   430→          type="button"
   431→        >
   432→          <Download className="size-3" />
   433→          Download
   434→        </Button>
   435→      </div>
   436→      <pre
   437→        ref={preRef}
   438→        className="max-h-96 overflow-auto rounded-md border bg-muted/30 p-3 font-mono text-xs leading-relaxed whitespace-pre-wrap break-words"
   439→      >
   440→        {logs}
   441→      </pre>
   442→    </div>
   443→  );
   444→}

   506→  const selectedId = useAppStore((s) => s.selectedId);
   507→  const inspectId = useAppStore((s) => s.inspectId);

   512→  const removeNode = useAppStore((s) => s.removeNode);
   513→  const select = useAppStore((s) => s.select);
   514→  const inspect = useAppStore((s) => s.inspect);
   515→  const setNodeStatus = useAppStore((s) => s.setNodeStatus);
   516→  const toast = useAppStore((s) => s.toast);
   517→
   518→  const patch = useDebouncedPatch();
   519→  const [running, setRunning] = React.useState(false);
   520→  const [runningAll, setRunningAll] = React.useState(false);
   521→  const [duplicating, setDuplicating] = React.useState(false);
   522→  const [viewerOpen, setViewerOpen] = React.useState(false);
   523→
   524→  const id = inspectId ?? selectedId;
   525→  const node = React.useMemo(
   526→    () => (workflow?.nodes ?? []).find((n) => n.id === id) ?? null,
   527→    [workflow, id],
   528→  );
   529→
   530→  const spec = React.useMemo(
   531→    () => (node ? nodeSpec(node.type) : undefined),
   532→    [node],
   533→  );
   534→
   535→  // Comp-tool nodes: parse the real output file list from the engine's
   536→  // ##OUTPUTS## trailer in the logs (paths under outputs/<tool>/<run>/).
   537→  const compToolKey =
   538→    node?.type && COMP_TOOLS.some((t) => t.key === node.type)
   539→      ? node.type
   540→      : node?.type === "comptool"
   541→        ? String(node.params.toolKey ?? "rfdiffusion")
   542→        : null;
   543→  const outputFiles = React.useMemo(() => {
   544→    if (!compToolKey || !node?.logs) return [];
   545→    const idx = node.logs.lastIndexOf("##OUTPUTS## ");
   546→    if (idx === -1) return [];
   547→    try {
   548→      const arr = JSON.parse(
   549→        node.logs.slice(idx + "##OUTPUTS## ".length).split("\n")[0],
   550→      ) as unknown;
   551→      return Array.isArray(arr) ? (arr as string[]) : [];
   552→    } catch {
   553→      return [];
   554→    }
   555→  }, [compToolKey, node?.logs]);
   556→  const viewerJob: ToolJobDTO | null = React.useMemo(() => {
   557→    if (!compToolKey || !node || outputFiles.length === 0) return null;
   558→    return {
   559→      id: `node-${node.id}`,
   560→      tool: compToolKey,
   561→      presetName: null,
   562→      params: {},
   563→      status: node.status === "failed" ? "failed" : "completed",
   564→      pid: null,
   565→      stdout: node.logs ?? "",
   566→      stderr: "",
   567→      outputFiles,
   568→      exitCode: node.status === "failed" ? 1 : 0,
   569→      command: node.logs?.split("\n")[0] ?? "",
   570→      triggeredBy: "workflow",
   571→      agentId: null,
   572→      environmentId: null,
   573→      startedAt: node.startedAt,
   574→      finishedAt: node.completedAt,
   575→      createdAt: node.createdAt,
   576→    };
   577→  }, [compToolKey, node, outputFiles]);
   578→
   579→  // ⌘+Enter / Ctrl+Enter to run this node. Ref holds the latest run function
   580→  // so the keyboard listener doesn't need to re-bind on every keystroke.
   581→  const runRef = React.useRef<(() => void) | null>(null);
   582→  React.useEffect(() => {
   583→    const onKey = (e: KeyboardEvent) => {
   584→      if ((e.metaKey || e.ctrlKey) && e.key === "Enter") {
   585→        e.preventDefault();
   586→        runRef.current?.();
   587→      }
   588→    };
   589→    window.addEventListener("keydown", onKey);
   590→    return () => window.removeEventListener("keydown", onKey);
   591→  }, []);
   592→
   593→  // Live progress via SSE — open an EventSource whenever the inspected node
   594→  // is running or pending. The server polls the DB every 500ms and emits a
   595→  // "status" event per poll, plus a terminal "done" event when the node
   596→  // reaches a completed/failed state. We surface each update in the store so
   597→  // the Logs/Result tabs + progress bar all animate live. The "done" toast
   598→  // is fired here (the onRun POST's own success toast was removed to avoid
   599→  // duplicates) — we also toast inside the status handler for terminal
   600→  // states, guarded by a `finished` flag, because the effect's cleanup may
   601→  // close the EventSource before the "done" event arrives (zustand updates
   602→  // synchronously, which re-renders and re-runs the effect, calling es.close()
   603→  // before the next SSE event tick).
   604→  React.useEffect(() => {
   605→    if (!node) return;
   606→    if (node.status !== "running" && node.status !== "pending") return;
   607→
   608→    let finished = false;
   609→    const nodeId = node.id;
   610→    const es = new EventSource(`/api/workflow/nodes/${nodeId}/stream`);
   611→
   612→    es.addEventListener("status", (e) => {
   613→      try {
   614→        const data = JSON.parse((e as MessageEvent).data) as {
   615→          status: NodeDTO["status"];
   616→          progress?: number;
   617→          logs?: string;
   618→          result?: string;
   619→        };
   620→        setNodeStatus(
   621→          nodeId,
   622→          data.status,
   623→          data.progress,
   624→          data.result,
   625→          data.logs,
   626→        );
   627→        if (data.status === "completed" || data.status === "failed") {
   628→          if (!finished) {
   629→            finished = true;
   630→            toast({
   631→              title: "Node finished",
   632→              description: data.status,
   633→              variant: data.status === "failed" ? "destructive" : "success",
   634→            });
   635→            es.close();
   636→          }
   637→        }
   638→      } catch {
   639→        // Malformed payload — ignore this tick.
   640→      }
   641→    });
   642→
   643→    es.addEventListener("done", (e) => {
   644→      try {
   645→        const data = JSON.parse((e as MessageEvent).data) as { status?: string };
   646→        if (!finished) {
   647→          finished = true;
   648→          toast({
   649→            title: "Node finished",
   650→            description: data.status ?? "completed",
   651→            variant: data.status === "failed" ? "destructive" : "success",
   652→          });
   653→        }
   654→      } catch {
   655→        // ignore
   656→      }
   657→      es.close();
   658→    });
   659→
   660→    es.addEventListener("error", (e) => {
   661→      // The native EventSource error event has no `data`; our backend's
   662→      // custom `event: error` frame does. Distinguish via `e.data`.
   663→      const ev = e as MessageEvent;
   664→      if (ev.data) {
   665→        try {
   666→          const data = JSON.parse(ev.data) as { error?: string };
   667→          if (data.error) {
   668→            toast({
   669→              title: "Stream error",
   670→              description: data.error,
   671→              variant: "destructive",
   672→            });
   673→          }
   674→        } catch {
   675→          // ignore
   676→        }
   677→      }
   678→      es.close();
   679→    });
   680→
   681→    return () => {
   682→      es.close();
   683→    };
   684→  }, [node?.id, node?.status]);
   685→
   686→  if (!node || !spec) return null;
   687→
   688→  const color = NODE_COLORS[spec.color as keyof typeof NODE_COLORS] ?? NODE_COLORS.slate;
   689→  const isRunning = node.status === "running" || running;
   690→
   691→  const onRename = (name: string) => {
   692→    upsertNode({ ...node, name });
   693→    patch(node.id, { name });
   694→  };
   695→
   696→  const onPatchParam = (key: string, value: string | number | boolean) => {
   697→    // For agent nodes, refId is a top-level field (not in params JSON).
   698→    if (node.type === "agent" && key === "refId") {
   699→      const updated = { ...node, refId: String(value) || null };
   700→      upsertNode(updated);
   701→      patch(node.id, { refId: String(value) || null });
   702→      return;
   703→    }
   704→    const params = { ...node.params, [key]: value };
   705→    upsertNode({ ...node, params });
   706→    patch(node.id, { params });
   707→  };
   708→
   709→  const onClose = () => {
   710→    select(null);
   711→    inspect(null);
   712→  };
   713→
   714→  const onRun = async () => {
   715→    if (isRunning) return;
   716→    setRunning(true);
   717→    setNodeStatus(node.id, "running", 10);
   718→    setTab("logs");
   719→    try {
   720→      const res = await fetch(`/api/workflow/nodes/${node.id}/run`, {
   721→        method: "POST",
   722→      });
   723→      if (!res.ok) {
   724→        const err = await res.json().catch(() => ({}));
   725→        throw new Error(err.error || `HTTP ${res.status}`);
   726→      }
   727→      const updated: NodeDTO = await res.json();
   728→      // The SSE effect above will toast "Node finished" when it observes the
   729→      // terminal status — no need to toast here too (would duplicate).
   730→      upsertNode(updated);
   731→    } catch (e) {
   732→      const msg = e instanceof Error ? e.message : String(e);
   733→      setNodeStatus(node.id, "failed", 100, `Error: ${msg}`, node.logs);
   734→      toast({ title: "Run failed", description: msg, variant: "destructive" });
   735→    } finally {
   736→      setRunning(false);
   737→    }
   738→  };
   739→  // Keep the ref in sync with the latest onRun closure.
   740→  runRef.current = onRun;
   741→
   742→  const onDuplicate = async () => {
   743→    if (duplicating) return;
   744→    setDuplicating(true);
   745→    try {
   746→      const res = await fetch("/api/workflow/nodes", {
   747→        method: "POST",
   748→        headers: { "Content-Type": "application/json" },
   749→        body: JSON.stringify({
   750→          type: node.type,
   751→          name: `${node.name} copy`,
   752→          x: node.x + 40,
   753→          y: node.y + 40,
   754→          refId: node.refId ?? undefined,
   755→          params: node.params,
   756→        }),
   757→      });
   758→      if (!res.ok) throw new Error("duplicate failed");
   759→      const created: NodeDTO = await res.json();
   760→      upsertNode(created);
   761→      useAppStore.getState().select(created.id);
   762→      useAppStore.getState().inspect(created.id);
   763→      toast({ title: "Duplicated", description: `${node.name} → ${created.name}` });
   764→    } catch {
   765→      toast({ title: "Duplicate failed", variant: "destructive" });
   766→    } finally {
   767→      setDuplicating(false);
   768→    }
   769→  };
   770→
   771→  const onRunAll = async () => {
   772→    setRunningAll(true);
   773→    try {
   774→      const res = await fetch("/api/workflow/run", { method: "POST" });
   775→      if (!res.ok) {
   776→        const err = await res.json().catch(() => ({}));
   777→        throw new Error(err.error || `HTTP ${res.status}`);
   778→      }
   779→      // Refetch the workflow to pull fresh node statuses.
   780→      const wfRes = await fetch("/api/workflow");
   781→      if (wfRes.ok) {
   782→        const wf = await wfRes.json();
   783→        useAppStore.getState().setWorkflow(wf);
   784→      }
   785→      toast({ title: "Workflow run complete", variant: "success" });
   786→    } catch (e) {
   787→      const msg = e instanceof Error ? e.message : String(e);
   788→      toast({ title: "Workflow run failed", description: msg, variant: "destructive" });
   789→    } finally {
   790→      setRunningAll(false);
   791→    }
   792→  };
   793→
   794→  const onDelete = async () => {
   795→    // Push history before removing the node.
   796→    const s = useAppStore.getState();
   797→    if (s.workflow) {
   798→      useHistoryStore.getState().push({
   799→        nodes: s.workflow.nodes,
   800→        edges: s.workflow.edges,
   801→        viewport: s.viewport,
   802→      });
   803→    }
   804→    try {
   805→      await fetch(`/api/workflow/nodes/${node.id}`, { method: "DELETE" });
   806→    } catch {
   807→      /* ignore — remove from local state anyway */
   808→    }
   809→    removeNode(node.id);
   810→    toast({ title: "Node deleted", description: node.name });
   811→  };
   812→
   813→  return (
   814→    <TooltipProvider delayDuration={250}>
   815→      <aside className="flex h-full w-80 shrink-0 flex-col border-l bg-background">
   816→        {/* Animated progress bar at the top while running */}
   817→        {isRunning && (
   818→          <div className="relative h-0.5 w-full overflow-hidden bg-muted">
   819→            <div
   820→              className="absolute inset-y-0 left-0 bg-primary transition-[width] duration-300"
   821→              style={{
   822→                width: `${Math.max(4, Math.min(100, node.progress || 10))}%`,
   823→              }}
   824→            />
   825→          </div>
   826→        )}
   827→
   828→        {/* Header with subtle gradient accent matching node color */}
   829→        <header
   830→          className={cn(
   831→            "relative flex flex-col gap-2 overflow-hidden border-b p-3",
   832→            "inspector-accent",
   833→          )}
   834→        >
   835→          {/* Colored top border accent */}
   836→          <div
   837→            className={cn(
   838→              "pointer-events-none absolute inset-x-0 top-0 h-0.5 opacity-70",
   839→              color.bg,
   840→            )}
   841→          />
   842→          <div className="flex items-center gap-2">
   843→            <span
   844→              className={cn(
   845→                "flex size-8 shrink-0 items-center justify-center rounded-md border",
   846→                color.soft,
   847→                color.border,
   848→                color.text,
   849→              )}
   850→            >
   851→              <SpecIcon name={spec.icon} className="size-4" />
   852→            </span>
   853→            <Input
   854→              value={node.name}
   855→              onChange={(e) => onRename(e.target.value)}
   856→              className="h-8 flex-1 text-sm font-medium"
   857→              aria-label="Node name"
   858→            />
   859→            <Button variant="ghost" size="icon" className="size-8" onClick={onClose} title="Close">
   860→              <X className="size-4" />
   861→            </Button>
   862→          </div>
   863→          <div className="flex items-center gap-2">
   864→            <Badge variant="outline" className={cn("border", STATUS_STYLES[node.status] ?? STATUS_STYLES.idle)}>
   865→              {node.status}
   866→            </Badge>
   867→            <span className="text-xs text-muted-foreground">{spec.label}</span>
   868→            {spec.usesLLM && (
   869→              <Badge variant="secondary" className="ml-auto text-[10px]">LLM</Badge>
   870→            )}
   871→          </div>
   872→        </header>
   873→
   874→        {/* Tabs */}
   875→        <Tabs value={tab} onValueChange={setTab} className="flex min-h-0 flex-1 flex-col">
   876→          <TabsList className="m-2 grid w-[calc(100%-1rem)] grid-cols-3">
   877→            <TabsTrigger value="params">Params</TabsTrigger>
   878→            <TabsTrigger value="logs">Logs</TabsTrigger>
   879→            <TabsTrigger value="result">Result</TabsTrigger>
   880→          </TabsList>
   881→          <ScrollArea className="min-h-0 flex-1">
   882→            <TabsContent value="params" className="m-0">
   883→              <ParamsTab
   884→                node={node}
   885→                spec={spec}
   886→                agents={agents}
   887→                onPatchParam={onPatchParam}
   888→              />
   889→            </TabsContent>
   890→            <TabsContent value="logs" className="m-0">
   891→              <LogsTab node={node} />
   892→            </TabsContent>
   893→            <TabsContent value="result" className="m-0">
   894→              <ResultTab node={node} />
   895→            </TabsContent>
   896→          </ScrollArea>
   897→        </Tabs>
   898→
   899→        <Separator />
   900→
   901→        {/* Footer actions */}
   902→        <footer className="flex flex-col gap-2 p-3">
   903→          <div className="flex gap-2">
   904→            <Tooltip>
   905→              <TooltipTrigger asChild>
   906→                <Button
   907→                  onClick={onRun}
   908→                  disabled={isRunning}
   909→                  className="flex-1"
   910→                  size="sm"
   911→                >
   912→                  {isRunning ? (
   913→                    <Loader2 className="size-4 animate-spin" />
   914→                  ) : (
   915→                    <Play className="size-4" />
   916→                  )}
   917→                  Run
   918→                </Button>
   919→              </TooltipTrigger>
   920→              <TooltipContent side="top">⌘+Enter to run</TooltipContent>
   921→            </Tooltip>
   922→            {viewerJob && (
   923→              <Tooltip>
   924→                <TooltipTrigger asChild>
   925→                  <Button
   926→                    onClick={() => setViewerOpen(true)}
   927→                    variant="outline"
   928→                    size="sm"
   929→                    className="gap-1"
   930→                  >
   931→                    <FileBox className="size-4" />
   932→                    Outputs
   933→                  </Button>
   934→                </TooltipTrigger>
   935→                <TooltipContent side="top">
   936→                  View {outputFiles.length} real output file(s)
   937→                </TooltipContent>
   938→              </Tooltip>
   939→            )}
   940→            <Tooltip>
   941→              <TooltipTrigger asChild>
   942→                <Button
   943→                  onClick={onDuplicate}
   944→                  disabled={duplicating}
   945→                  variant="outline"
   946→                  size="sm"
   947→                  className="gap-1.5"
   948→                >
   949→                  {duplicating ? (
   950→                    <Loader2 className="size-4 animate-spin" />
   951→                  ) : (
   952→                    <Copy className="size-4" />
   953→                  )}
   954→                  Duplicate
   955→                </Button>
   956→              </TooltipTrigger>
   957→              <TooltipContent side="top">Duplicate this node</TooltipContent>
   958→            </Tooltip>
   959→          </div>
   960→          <Button
   961→            onClick={onRunAll}
   962→            disabled={runningAll}
   963→            variant="secondary"
   964→            size="sm"
   965→            className="w-full"
   966→          >
   967→            {runningAll ? (
   968→              <Loader2 className="size-4 animate-spin" />
   969→            ) : (
   970→              <Play className="size-4" />
   971→            )}
   972→            Run All
   973→          </Button>
   974→          <AlertDialog>
   975→            <AlertDialogTrigger asChild>
   976→              <Button variant="ghost" size="sm" className="text-destructive hover:bg-destructive/10 hover:text-destructive">
   977→                <Trash2 className="size-4" />
   978→                Delete node
   979→              </Button>
   980→            </AlertDialogTrigger>
   981→            <AlertDialogContent>
   982→              <AlertDialogHeader>
   983→                <AlertDialogTitle>Delete “{node.name}”?</AlertDialogTitle>
   984→                <AlertDialogDescription>
   985→                  This removes the node and any connected edges. This cannot be undone.
   986→                </AlertDialogDescription>
   987→              </AlertDialogHeader>
   988→              <AlertDialogFooter>
   989→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   990→                <AlertDialogAction
   991→                  onClick={onDelete}
   992→                  className="bg-destructive text-white hover:bg-destructive/90"
   993→                >
   994→                  Delete
   995→                </AlertDialogAction>
   996→              </AlertDialogFooter>
   997→            </AlertDialogContent>
   998→          </AlertDialog>
   999→        </footer>
  1000→        {viewerJob && (
  1001→          <OutputViewerDialog
  1002→            job={viewerJob}
  1003→            open={viewerOpen}
  1004→            onClose={() => setViewerOpen(false)}
  1005→            nodeMode
  1006→          />
  1007→        )}
  1008→      </aside>
  1009→    </TooltipProvider>
  1010→  );
  1011→}
  1012→
  1013→export const NodeInspector = React.memo(NodeInspectorImpl);

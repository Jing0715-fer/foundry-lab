    44→function timeAgo(iso: string): string {
    45→  const then = new Date(iso).getTime();
    46→  const now = Date.now();
    47→  const s = Math.max(1, Math.round((now - then) / 1000));
    48→  if (s < 60) return `${s}s ago`;
    49→  const m = Math.round(s / 60);
    50→  if (m < 60) return `${m}m ago`;
    51→  const h = Math.round(m / 60);
    52→  if (h < 24) return `${h}h ago`;
    53→  const d = Math.round(h / 24);
    54→  return `${d}d ago`;
    55→}

    57→/**
    58→ * Detect whether a ToolJob's stdout indicates a simulated run. The
    59→ * real-executor (src/lib/real-executor.ts) prefixes simulated output with
    60→ * `[SIMULATED — ...]`; the legacy executor (executeCompTool →
    61→ * simulateCompRun) emits `(simulated)` in its first line and
    62→ * `FOUNDRY-LAB SIMULATION` in the PDB REMARK. We treat any of those as
    63→ * evidence the run was a simulation rather than a real algorithm.
    64→ */
    65→function isSimulatedJob(job: ToolJobDTO): boolean {
    66→  if (job.params && job.params.simulated === true) return true;
    67→  const out = job.stdout ?? "";
    68→  if (/^\[SIMULATED/i.test(out)) return true;
    69→  if (/\(simulated\)/i.test(out)) return true;
    70→  if (/FOUNDRY-LAB SIMULATION/i.test(out)) return true;
    71→  if (job.command && /^\[SIMULATED\]/i.test(job.command)) return true;
    72→  return false;
    73→}

   553→                const JobIcon = (def && COMP_TOOL_ICONS[def.icon]) || Wrench;

   555→                  OUTPUT_TYPE_META[job.tool] ?? OUTPUT_TYPE_FALLBACK;

   557→                const simulated = isSimulatedJob(job);
   558→                const isActive =
   559→                  job.status === "running" ||
   560→                  job.status === "pending" ||
   561→                  job.status === "queued";
   562→                const statusPill =
   563→                  job.status === "completed"
   564→                    ? "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   565→                    : job.status === "failed"
   566→                      ? "bg-rose-500/10 text-rose-700 dark:text-rose-400"
   567→                      : isActive
   568→                        ? "bg-amber-500/10 text-amber-700 dark:text-amber-400"
   569→                        : "bg-muted text-muted-foreground";
   570→                const isExpanded = expandedJobId === job.id;
   571→                return (
   572→                  <li
   573→                    key={job.id}
   574→                    className={`rounded-md border transition-colors ${
   575→                      isExpanded ? "border-primary/40" : ""
   576→                    }`}
   577→                  >
   578→                    <div className="flex flex-wrap items-center gap-2 p-2.5 text-sm">
   579→                      <button
   580→                        type="button"
   581→                        onClick={() =>
   582→                          setExpandedJobId((id) => (id === job.id ? null : job.id))
   583→                        }
   584→                        className="flex size-6 items-center justify-center rounded-md bg-muted text-muted-foreground hover:bg-accent"
   585→                        title={isExpanded ? "Hide preview" : "Show preview"}
   586→                        aria-label={isExpanded ? "Hide preview" : "Show preview"}
   587→                      >
   588→                        {isExpanded ? (
   589→                          <ChevronDown className="size-3.5" />
   590→                        ) : (
   591→                          <ChevronRight className="size-3.5" />
   592→                        )}
   593→                      </button>
   594→                      <span className="flex size-6 items-center justify-center rounded-md bg-muted text-muted-foreground">
   595→                        <JobIcon className="size-3.5" />
   596→                      </span>
   597→                      <span
   598→                        className={`flex size-5 items-center justify-center rounded-[4px] ${outputMeta.chip}`}
   599→                        title={`Output type: ${outputMeta.label}`}
   600→                      >
   601→                        <OutputIcon className="size-3" />
   602→                      </span>
   603→                      <Badge variant="secondary" className="font-mono text-[10px]">
   604→                        {job.tool}
   605→                      </Badge>
   606→                      <span
   607→                        className={`inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-[10px] font-medium ${statusPill}`}
   608→                      >
   609→                        {isActive && <Loader2 className="size-2.5 animate-spin" />}
   610→                        {job.status}
   611→                      </span>
   612→                      {/* REAL / SIMULATED badge — surfaces whether the run
   613→                          used a real algorithm or fell back to the
   614→                          synthetic generator. */}
   615→                      <Badge
   616→                        variant="outline"
   617→                        className={
   618→                          simulated
   619→                            ? "border-amber-500/40 bg-amber-500/10 text-[10px] text-amber-700 dark:text-amber-400"
   620→                            : "border-emerald-500/40 bg-emerald-500/10 text-[10px] text-emerald-700 dark:text-emerald-400"
   621→                        }
   622→                        title={
   623→                          simulated
   624→                            ? "This run used the synthetic generator (no real tool was installed)"
   625→                            : "This run used a real algorithm"
   626→                        }
   627→                      >
   628→                        {simulated ? "SIMULATED" : "REAL"}
   629→                      </Badge>
   630→                      <span className="text-xs text-muted-foreground">
   631→                        by {job.triggeredBy || "user"}
   632→                      </span>
   633→                      <span className="ml-auto flex items-center gap-1 text-xs text-muted-foreground">
   634→                        <Clock className="size-3" />
   635→                        {timeAgo(job.createdAt)}
   636→                      </span>
   637→                      <Button
   638→                        size="sm"
   639→                        variant="outline"
   640→                        onClick={() => setViewJob(job)}
   641→                      >
   642→                        <Eye className="size-3.5" />
   643→                        View output
   644→                      </Button>
   645→                    </div>
   646→                    {/* Inline preview — collapsible stdout + per-file
   647→                        download buttons. Shown when the row is expanded so
   648→                        users can scan results without opening the dialog. */}
   649→                    {isExpanded && (
   650→                      <div className="space-y-3 border-t bg-muted/20 px-3 py-3">
   651→                        {job.stdout ? (
   652→                          <div>
   653→                            <p className="mb-1 text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   654→                              stdout
   655→                            </p>
   656→                            <pre className="max-h-64 overflow-auto rounded-md bg-background p-3 font-mono text-xs leading-relaxed whitespace-pre-wrap break-words">
   657→                              {job.stdout}
   658→                            </pre>
   659→                          </div>
   660→                        ) : (
   661→                          <p className="text-xs text-muted-foreground">
   662→                            No stdout captured for this job.
   663→                          </p>
   664→                        )}
   665→                        {job.stderr && (
   666→                          <div>
   667→                            <p className="mb-1 text-[10px] font-medium uppercase tracking-wide text-destructive">
   668→                              stderr
   669→                            </p>
   670→                            <pre className="max-h-32 overflow-auto rounded-md bg-destructive/10 p-3 font-mono text-xs text-destructive whitespace-pre-wrap break-words">
   671→                              {job.stderr}
   672→                            </pre>
   673→                          </div>
   674→                        )}
   675→                        {job.outputFiles.length > 0 && (
   676→                          <div>
   677→                            <p className="mb-1.5 text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   678→                              output files ({job.outputFiles.length})
   679→                            </p>
   680→                            <ul className="flex flex-wrap gap-1.5">
   681→                              {job.outputFiles.map((f, i) => (
   682→                                <li key={`${f}-${i}`}>
   683→                                  <Button
   684→                                    size="sm"
   685→                                    variant="outline"
   686→                                    asChild
   687→                                    className="h-7 gap-1 px-2 text-[11px]"
   688→                                  >
   689→                                    <a
   690→                                      href={fileDownloadUrl(job.id, f)}
   691→                                      target="_blank"
   692→                                      rel="noopener noreferrer"
   693→                                    >
   694→                                      <Download className="size-3" />
   695→                                      <code className="font-mono">
   696→                                        {f.split("/").pop() ?? f}
   697→                                      </code>
   698→                                    </a>
   699→                                  </Button>
   700→                                </li>
   701→                              ))}
   702→                            </ul>
   703→                          </div>
   704→                        )}
   705→                      </div>
   706→                    )}
   707→                  </li>
   708→                );
   709→              })}
   710→            </ul>
   711→          )}
   712→        </CardContent>
   713→      </Card>
   714→
   715→      {/* Job output dialog (tabbed: Summary / Structure / Sequence / Files / Command) */}
   716→      <OutputViewerDialog
   717→        job={viewJob}
   718→        open={!!viewJob}
   719→        onClose={() => setViewJob(null)}
   720→      />
   721→    </div>
   722→  );
   723→}
   724→
   725→function ParamField({
   726→  field,
   727→  value,
   728→  onChange,
   729→}: {
   730→  field: CompParamField;
   731→  value: unknown;
   732→  onChange: (v: unknown) => void;
   733→}) {
   734→  const labelEl = (
   735→    <Label className="text-xs">
   736→      {field.label}
   737→      {field.unit && <span className="ml-1 text-muted-foreground">({field.unit})</span>}
   738→    </Label>
   739→  );
   740→
   741→  if (field.type === "bool") {
   742→    return (
   743→      <div className="flex items-center justify-between rounded-md border p-2.5">
   744→        <div>
   745→          {labelEl}
   746→          {field.hint && <p className="text-[10px] text-muted-foreground">{field.hint}</p>}
   747→        </div>
   748→        <Switch checked={!!value} onCheckedChange={(v) => onChange(v)} />
   749→      </div>
   750→    );
   751→  }
   752→
   753→  if (field.type === "select") {
   754→    return (
   755→      <div className="grid gap-1.5">
   756→        {labelEl}
   757→        <Select value={String(value)} onValueChange={(v) => onChange(v)}>
   758→          <SelectTrigger className="w-full">
   759→            <SelectValue />
   760→          </SelectTrigger>
   761→          <SelectContent>
   762→            {(field.options ?? []).map((opt) => (
   763→              <SelectItem key={opt} value={opt}>
   764→                {opt}
   765→              </SelectItem>
   766→            ))}
   767→          </SelectContent>
   768→        </Select>
   769→      </div>
   770→    );
   771→  }
   772→
   773→  if (field.type === "number") {
   774→    return (
   775→      <div className="grid gap-1.5">
   776→        {labelEl}
   777→        <Input
   778→          type="number"
   779→          min={field.min}
   780→          max={field.max}
   781→          step={field.step ?? 1}
   782→          value={value as number}
   783→          onChange={(e) => onChange(Number(e.target.value))}
   784→        />
   785→        {field.hint && <p className="text-[10px] text-muted-foreground">{field.hint}</p>}
   786→      </div>
   787→    );
   788→  }
   789→
   790→  // text | path
   791→  return (
   792→    <div className="grid gap-1.5">
   793→      {labelEl}
   794→      <Input
   795→        type="text"
   796→        value={String(value ?? "")}
   797→        onChange={(e) => onChange(e.target.value)}
   798→        placeholder={field.hint ?? ""}
   799→      />
   800→      {field.hint && <p className="text-[10px] text-muted-foreground">{field.hint}</p>}
   801→    </div>
   802→  );
   803→}
